(

    SHIFT_TO_PROCESSING_AREA := 2,  // Steps 2-11: Position block for cutting
                                    //   Axis 1 shifts block
                                    //   Axes 2,3,4,5 raise/lower
                                    //   Axes 6,7 shift to center
                                    //   Axes 8,9 shift to center
                                    //   Axes 10,11,12,13 raise/lower
                                    //   Suction cups engage

    CUTTING := 3,                   // G-CODE DRIVEN CUTTING
                                    // Replaces: STUDS, WINDOWS_DOORS, ELECTRICAL
                                    // Python writes cutting path to GVL_GCode.aMoveQueue
                                    // PLC executes moves on hot wire axes:
                                    //   Y (vertical):   14,18,22,26 (+ slaves 16,20,24,28)
                                    //   X (horizontal): 15,19,23,27 (+ slaves 17,21,25,29)
                                    // Handles all cutting operations:
                                    //   - Stud carving
                                    //   - Window/door openings
                                    //   - Electrical chases
                                    // Skip this step if GVL_GCode.nQueueCount = 0

    SHIFT_TO_UNLOAD_AREA := 4       // Steps: Move block to unload position
                                    //   Axes 10,11,12,13 raise
                                    //   Axes 8,9 shift block to unload
                                    //   Axes 10,11,12,13 lower
                                    //   Axes 8,9 return home
);
END_TYPE
