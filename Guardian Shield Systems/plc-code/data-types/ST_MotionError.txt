TYPE ST_MotionError :
// Error information structure for motion faults
STRUCT
    hasError : BOOL;            // TRUE if an error exists
    errorAxisID : UINT;         // Which axis caused the error (0 = none)
    errorCode : DWORD;          // Application error code (1001=Power, 2001=LoadBlock, 2002=Cutting, 2005=Unload)
    mcErrorID : DWORD;          // Raw PLCopen MC error code from MC_MoveAbsolute (0 = N/A for cutting/power errors)
    errorStep : E_MotionStep;   // Which motion step the error occurred in
    errorSubStep : UINT;         // Which substep the error occurred in
    errorTimestamp : TIME;      // When the error occurred
    errorDescription : STRING[80]; // Human-readable error description
END_STRUCT
END_TYPE
