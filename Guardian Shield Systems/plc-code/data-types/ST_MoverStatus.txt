TYPE ST_MoverStatus :
// ============================================================================
// Lightweight structure to capture FB_AxisMover status for aggregation
// Used by FC_CheckMoversComplete and FC_FindFirstErrorAxis utilities
//
// Usage:
//   status1 := (Done := moverAxis14.Done, Error := moverAxis14.Error, AxisID := 14);
//   errorAxis := FC_FindFirstErrorAxis(status1, status2, status3, ...);
// ============================================================================
STRUCT
    Done    : BOOL;     // Mover has completed its move
    Error   : BOOL;     // Mover has encountered an error
    AxisID  : UINT;     // Axis number this status represents (1-36)
END_STRUCT
END_TYPE
