(*
==============================================================================
  ST_ScriptCmd
  Input payload for ctrlX script/instances/{name}/cmd/file Data Layer call.

  Maps to the JSON body:
    { "name": "/path/to/file.npg", "param": [] }

  Used by FB_ScriptDispatch when calling DL_WriteNode to dispatch a G-code
  file to a script instance.
==============================================================================
*)

TYPE ST_ScriptCmd :
STRUCT
    name  : STRING(255);                    // Full path to .npg file on ctrlX
    param : ARRAY[0..0] OF STRING(1);       // Required by API — always empty
END_STRUCT
END_TYPE
