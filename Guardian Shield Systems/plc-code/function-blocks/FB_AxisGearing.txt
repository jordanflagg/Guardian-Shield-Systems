{attribute 'no_assign'}
FUNCTION_BLOCK FB_AxisGearing
// ============================================================================
// Manages all permanent gearing relationships between master and slave axes.
// Due to VAR_IN_OUT requirements of MB_GearInPos, each pair requires explicit calls.
// Call this function block to establish/re-establish gearing after any
// interruption (MC_Halt, E-Stop, fault recovery, etc.)
//
// Gearing Map (18 pairs total):
//   Load Table:    2 -> 3, 4, 5
//   Shift Center:  6 -> 7
//   Pusher:        8 -> 9
//   Unload Table:  10 -> 11, 12, 13
//   Studs:         14->16, 15->17, 18->20, 19->21, 22->24, 23->25, 26->28, 27->29
//   Window/Door:   30->32, 31->33
// ============================================================================

VAR_INPUT
    Execute : BOOL;             // TRUE to engage gearing, FALSE to disengage
END_VAR

VAR_OUTPUT
    Done : BOOL;                // All gearing successfully engaged
    Busy : BOOL;                // Gearing in progress
    Error : BOOL;               // One or more gearing operations failed
    ErrorAxisID : UINT;         // Which slave axis had the error (0 = none)
    ActivePairCount : UINT;     // Number of gearing pairs currently active
END_VAR

VAR CONSTANT
    NUM_GEARING_PAIRS : UINT := 18;
END_VAR

VAR
    // =========================================================================
    // Gearing Function Blocks (MB_GearInPos requires VAR_IN_OUT for axes)
    // Organized by machine section for clarity
    // =========================================================================

    // Load table lift: Axis 2 master -> slaves 3, 4, 5
    fbGear_2_3 : MB_GearInPos;
    fbGear_2_4 : MB_GearInPos;
    fbGear_2_5 : MB_GearInPos;

    // Shift to center: Axis 6 master -> slave 7
    fbGear_6_7 : MB_GearInPos;

    // Unload pusher: Axis 8 master -> slave 9
    fbGear_8_9 : MB_GearInPos;

    // Unload table lift: Axis 10 master -> slaves 11, 12, 13
    fbGear_10_11 : MB_GearInPos;
    fbGear_10_12 : MB_GearInPos;
    fbGear_10_13 : MB_GearInPos;

    // Stud carving (8 pairs)
    fbGear_14_16 : MB_GearInPos;
    fbGear_15_17 : MB_GearInPos;
    fbGear_18_20 : MB_GearInPos;
    fbGear_19_21 : MB_GearInPos;
    fbGear_22_24 : MB_GearInPos;
    fbGear_23_25 : MB_GearInPos;
    fbGear_26_28 : MB_GearInPos;
    fbGear_27_29 : MB_GearInPos;

    // Window/door hot wire (2 pairs)
    fbGear_30_32 : MB_GearInPos;
    fbGear_31_33 : MB_GearInPos;

    // =========================================================================
    // Status Aggregation Arrays
    // Maps pair index to slave axis ID for error reporting
    // =========================================================================
    SLAVE_AXIS_MAP : ARRAY[1..18] OF UINT := [
        3,   // Pair 1:  2->3
        4,   // Pair 2:  2->4
        5,   // Pair 3:  2->5
        7,   // Pair 4:  6->7
        9,   // Pair 5:  8->9
        11,  // Pair 6:  10->11
        12,  // Pair 7:  10->12
        13,  // Pair 8:  10->13
        16,  // Pair 9:  14->16
        17,  // Pair 10: 15->17
        20,  // Pair 11: 18->20
        21,  // Pair 12: 19->21
        24,  // Pair 13: 22->24
        25,  // Pair 14: 23->25
        28,  // Pair 15: 26->28
        29,  // Pair 16: 27->29
        32,  // Pair 17: 30->32
        33   // Pair 18: 31->33
    ];

    // Collected status from all gearing FBs (populated each scan)
    pairInSync : ARRAY[1..18] OF BOOL;
    pairError : ARRAY[1..18] OF BOOL;

    // Edge detection
    executePrev : BOOL := FALSE;
    executeRising : BOOL := FALSE;

    // State tracking
    gearingActive : BOOL := FALSE;

    // Loop counter
    i : UINT;
END_VAR

///////////////////////////////////////////////////////////////////////////////

// Edge detection for Execute
executeRising := Execute AND NOT executePrev;
executePrev := Execute;

// ============================================================================
// DISENGAGE - When Execute goes FALSE
// ============================================================================
IF NOT Execute THEN
    Done := FALSE;
    Busy := FALSE;
    Error := FALSE;
    ErrorAxisID := 0;
    ActivePairCount := 0;
    gearingActive := FALSE;

    // Disengage all gearing FBs (must provide axis references for VAR_IN_OUT)
    fbGear_2_3(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[2], Slave := AXIF_CONFIG_INDEXES[3]);
    fbGear_2_4(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[2], Slave := AXIF_CONFIG_INDEXES[4]);
    fbGear_2_5(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[2], Slave := AXIF_CONFIG_INDEXES[5]);
    fbGear_6_7(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[6], Slave := AXIF_CONFIG_INDEXES[7]);
    fbGear_8_9(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[8], Slave := AXIF_CONFIG_INDEXES[9]);
    fbGear_10_11(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[10], Slave := AXIF_CONFIG_INDEXES[11]);
    fbGear_10_12(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[10], Slave := AXIF_CONFIG_INDEXES[12]);
    fbGear_10_13(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[10], Slave := AXIF_CONFIG_INDEXES[13]);
    fbGear_14_16(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[14], Slave := AXIF_CONFIG_INDEXES[16]);
    fbGear_15_17(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[15], Slave := AXIF_CONFIG_INDEXES[17]);
    fbGear_18_20(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[18], Slave := AXIF_CONFIG_INDEXES[20]);
    fbGear_19_21(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[19], Slave := AXIF_CONFIG_INDEXES[21]);
    fbGear_22_24(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[22], Slave := AXIF_CONFIG_INDEXES[24]);
    fbGear_23_25(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[23], Slave := AXIF_CONFIG_INDEXES[25]);
    fbGear_26_28(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[26], Slave := AXIF_CONFIG_INDEXES[28]);
    fbGear_27_29(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[27], Slave := AXIF_CONFIG_INDEXES[29]);
    fbGear_30_32(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[30], Slave := AXIF_CONFIG_INDEXES[32]);
    fbGear_31_33(Execute := FALSE, Master := AXIF_CONFIG_INDEXES[31], Slave := AXIF_CONFIG_INDEXES[33]);

    RETURN;
END_IF

// ============================================================================
// START - On rising edge of Execute
// ============================================================================
IF executeRising THEN
    Busy := TRUE;
    Done := FALSE;
    Error := FALSE;
    ErrorAxisID := 0;
    gearingActive := TRUE;
END_IF

// ============================================================================
// ENGAGE - Call all gearing FBs with 1:1 ratio
// Each pair requires explicit call due to VAR_IN_OUT requirements
// ============================================================================

// Load table lift: Axis 2 master -> slaves 3, 4, 5
fbGear_2_3(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[2], Slave := AXIF_CONFIG_INDEXES[3],
           RatioNumerator := 1, RatioDenominator := 1);
fbGear_2_4(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[2], Slave := AXIF_CONFIG_INDEXES[4],
           RatioNumerator := 1, RatioDenominator := 1);
fbGear_2_5(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[2], Slave := AXIF_CONFIG_INDEXES[5],
           RatioNumerator := 1, RatioDenominator := 1);

// Shift to center: Axis 6 master -> slave 7
fbGear_6_7(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[6], Slave := AXIF_CONFIG_INDEXES[7],
           RatioNumerator := 1, RatioDenominator := 1);

// Unload pusher: Axis 8 master -> slave 9
fbGear_8_9(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[8], Slave := AXIF_CONFIG_INDEXES[9],
           RatioNumerator := 1, RatioDenominator := 1);

// Unload table lift: Axis 10 master -> slaves 11, 12, 13
fbGear_10_11(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[10], Slave := AXIF_CONFIG_INDEXES[11],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_10_12(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[10], Slave := AXIF_CONFIG_INDEXES[12],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_10_13(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[10], Slave := AXIF_CONFIG_INDEXES[13],
             RatioNumerator := 1, RatioDenominator := 1);

// Stud carving pairs (8 pairs)
fbGear_14_16(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[14], Slave := AXIF_CONFIG_INDEXES[16],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_15_17(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[15], Slave := AXIF_CONFIG_INDEXES[17],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_18_20(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[18], Slave := AXIF_CONFIG_INDEXES[20],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_19_21(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[19], Slave := AXIF_CONFIG_INDEXES[21],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_22_24(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[22], Slave := AXIF_CONFIG_INDEXES[24],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_23_25(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[23], Slave := AXIF_CONFIG_INDEXES[25],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_26_28(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[26], Slave := AXIF_CONFIG_INDEXES[28],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_27_29(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[27], Slave := AXIF_CONFIG_INDEXES[29],
             RatioNumerator := 1, RatioDenominator := 1);

// Window/door hot wire (2 pairs)
fbGear_30_32(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[30], Slave := AXIF_CONFIG_INDEXES[32],
             RatioNumerator := 1, RatioDenominator := 1);
fbGear_31_33(Execute := gearingActive, Master := AXIF_CONFIG_INDEXES[31], Slave := AXIF_CONFIG_INDEXES[33],
             RatioNumerator := 1, RatioDenominator := 1);

// ============================================================================
// COLLECT STATUS - Gather InSync and Error from all FBs into arrays
// ============================================================================
pairInSync[1]  := fbGear_2_3.InSync;   pairError[1]  := fbGear_2_3.Error;
pairInSync[2]  := fbGear_2_4.InSync;   pairError[2]  := fbGear_2_4.Error;
pairInSync[3]  := fbGear_2_5.InSync;   pairError[3]  := fbGear_2_5.Error;
pairInSync[4]  := fbGear_6_7.InSync;   pairError[4]  := fbGear_6_7.Error;
pairInSync[5]  := fbGear_8_9.InSync;   pairError[5]  := fbGear_8_9.Error;
pairInSync[6]  := fbGear_10_11.InSync; pairError[6]  := fbGear_10_11.Error;
pairInSync[7]  := fbGear_10_12.InSync; pairError[7]  := fbGear_10_12.Error;
pairInSync[8]  := fbGear_10_13.InSync; pairError[8]  := fbGear_10_13.Error;
pairInSync[9]  := fbGear_14_16.InSync; pairError[9]  := fbGear_14_16.Error;
pairInSync[10] := fbGear_15_17.InSync; pairError[10] := fbGear_15_17.Error;
pairInSync[11] := fbGear_18_20.InSync; pairError[11] := fbGear_18_20.Error;
pairInSync[12] := fbGear_19_21.InSync; pairError[12] := fbGear_19_21.Error;
pairInSync[13] := fbGear_22_24.InSync; pairError[13] := fbGear_22_24.Error;
pairInSync[14] := fbGear_23_25.InSync; pairError[14] := fbGear_23_25.Error;
pairInSync[15] := fbGear_26_28.InSync; pairError[15] := fbGear_26_28.Error;
pairInSync[16] := fbGear_27_29.InSync; pairError[16] := fbGear_27_29.Error;
pairInSync[17] := fbGear_30_32.InSync; pairError[17] := fbGear_30_32.Error;
pairInSync[18] := fbGear_31_33.InSync; pairError[18] := fbGear_31_33.Error;

// ============================================================================
// AGGREGATE STATUS - Loop through arrays to determine overall state
// ============================================================================
IF gearingActive THEN
    ActivePairCount := 0;
    Error := FALSE;
    ErrorAxisID := 0;
    Done := TRUE;  // Assume done, clear if any pair not in sync

    FOR i := 1 TO NUM_GEARING_PAIRS DO
        // Check for first error (report slave axis that failed)
        IF pairError[i] AND NOT Error THEN
            Error := TRUE;
            ErrorAxisID := SLAVE_AXIS_MAP[i];
        END_IF

        // Count pairs in sync
        IF pairInSync[i] THEN
            ActivePairCount := ActivePairCount + 1;
        ELSE
            Done := FALSE;  // Not all pairs in sync yet
        END_IF
    END_FOR

    // Handle error condition
    IF Error THEN
        Busy := FALSE;
        Done := FALSE;
        gearingActive := FALSE;
    ELSIF Done THEN
        // All pairs in sync - gearing complete
        Busy := FALSE;
        // Keep gearingActive TRUE to maintain gearing
    END_IF
END_IF
