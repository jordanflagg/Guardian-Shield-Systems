FUNCTION_BLOCK FB_AxisHome
(*
==============================================================================
  FB_AxisHome — Homing for all 18 master axes
  "Home" means moving to position 0 (all axes use absolute encoders).

  AXIS CATEGORIES:
  ─────────────────
  Non-kinematic masters — free to command with MC_MoveAbsolute at any time:
    Axis 1, 2, 6, 8, 10, 34

  Kinematic masters — inside always-enabled kin groups (Wire1-4, Wire5_6).
  MC_MoveAbsolute will fault while the kin is enabled. Homed by dispatching
  Wire*_home.npg scripts via trigger_dispatch.py:
    Wire1-4 kin : Axis 14, 15, 18, 19, 22, 23, 26, 27
    Wire5_6 kin : Axis 30, 31, 35, 36

  HOME ALL SEQUENCE:
    1. Home 6 non-kin masters in parallel (one FB_MultiAxisMover call).
    2. Set GVL_GCode.bHomeRequest = TRUE.
       trigger_dispatch.py sees this, dispatches Wire1-4 home scripts
       simultaneously, then Wire5_6 home script, then sets bHomeDone = TRUE.
    3. Wait for GVL_GCode.bHomeDone, then clear both flags.

  HOME SINGLE:
    - Non-kin axis: MC_MoveAbsolute to 0.
    - Kin axis: triggers bHomeRequest (homes all kin axes in both groups).
      Individual kin axis homing is not possible without disabling the group.
==============================================================================
*)

VAR_INPUT
    Enable : BOOL;                  // Enable homing mode
    HomeAxis : UINT;                // Axis to home (1-36 for single, 0 for all)
    HomeSingle : BOOL;              // Rising edge: home selected axis
    HomeAll : BOOL;                 // Rising edge: home all master axes
    Velocity : LREAL := 10.0;       // Homing velocity mm/s (non-kin axes only)
    Acceleration : LREAL := 5.0;
    Deceleration : LREAL := 5.0;
    Jerk : LREAL := 10.0;
    PositionTolerance : LREAL := 0.5; // Tolerance for "at home" check (mm)
    Abort : BOOL;                     // Halt all in-progress moves immediately
END_VAR

VAR_OUTPUT
    Busy : BOOL;                    // Homing operation in progress
    Done : BOOL;                    // Homing operation completed
    Error : BOOL;                   // Homing error occurred
    ErrorAxisID : UINT;             // Axis that caused error
    AllAxesHomed : BOOL;            // TRUE when all master axes are at position 0
    AxisIsHomed : ARRAY[1..36] OF BOOL := [36(TRUE)];
    InvalidAxis : BOOL;             // Selected axis is not a master axis
    Aborted : BOOL;                 // TRUE after a clean abort/halt sequence completes
END_VAR

VAR
    // Non-kin master mover (axes 1, 2, 6, 8, 10, 34 — up to 6 in one group)
    fbMoveNonKin : FB_MultiAxisMover;

    // Single axis mover for individual non-kin home commands
    fbMoveSingle : FB_AxisMover;

    // State machine
    state : UINT := 0;
    homeSinglePrev : BOOL;
    homeAllPrev : BOOL;
    homeSingleRising : BOOL;
    homeAllRising : BOOL;

    // Tracking
    selectedAxis : UINT;
    selectedAxisRef : AXIS_REF;
    selectedAxisIsKin : BOOL;   // TRUE when selected axis is inside a kin group

    // Master axis validation
    isMaster : BOOL;
    i : UINT;

    // Position reading (one entry per master axis, indexed 1..18)
    fbReadPos : ARRAY[1..18] OF MC_ReadActualPosition;
    currentPos : LREAL;

    // Abort/halt support
    fbHalt      : ARRAY[1..6] OF MC_Halt;
    haltDone    : ARRAY[1..6] OF BOOL;
    haltActive  : BOOL;
    haltAllDone : BOOL;
END_VAR

// ─── Kinematic axis lookup ────────────────────────────────────────────────────
// Axes inside always-enabled kins — cannot use MC_MoveAbsolute while kin is active.
// Wire1-4 kin: 14, 15, 18, 19, 22, 23, 26, 27
// Wire5_6 kin: 30, 31, 35, 36
///////////////////////////////////////////////////////////////////////////////

// Edge detection
homeSingleRising := HomeSingle AND NOT homeSinglePrev;
homeSinglePrev := HomeSingle;
homeAllRising := HomeAll AND NOT homeAllPrev;
homeAllPrev := HomeAll;

// Validate selected axis
isMaster := FALSE;
FOR i := 1 TO GVL_Axes.MASTER_AXIS_COUNT DO
    IF HomeAxis = GVL_Axes.MASTER_AXES[i] THEN
        isMaster := TRUE;
        EXIT;
    END_IF
END_FOR
InvalidAxis := (HomeAxis > 0) AND NOT isMaster;

// Check if selected axis is inside a kinematic group
selectedAxisIsKin := FALSE;
FOR i := 1 TO GVL_Axes.KIN_MASTER_COUNT DO
    IF HomeAxis = GVL_Axes.KIN_MASTER_AXES[i] THEN
        selectedAxisIsKin := TRUE;
        EXIT;
    END_IF
END_FOR

selectedAxisRef := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[1]];
IF selectedAxis >= MOTIF_CONFIG.MIN_AXIS_INDEX AND selectedAxis <= MOTIF_CONFIG.MAX_AXIS_INDEX THEN
    selectedAxisRef := AXIF_CONFIG_INDEXES[selectedAxis];
END_IF

// Read MASTER axis positions always — slaves follow via gantry, never addressed directly.
// Position reading works even with kins enabled.
FOR i := 1 TO GVL_Axes.MASTER_AXIS_COUNT DO
    fbReadPos[i](
        Enable := TRUE,
        Axis := AXIF_CONFIG_INDEXES[GVL_Axes.MASTER_AXES[i]]
    );
    IF fbReadPos[i].Valid THEN
        currentPos := fbReadPos[i].Position;
        AxisIsHomed[GVL_Axes.MASTER_AXES[i]] := ABS(currentPos) <= PositionTolerance;
    END_IF
END_FOR

// Check if all master axes are homed
AllAxesHomed := TRUE;
FOR i := 1 TO GVL_Axes.MASTER_AXIS_COUNT DO
    IF NOT AxisIsHomed[GVL_Axes.MASTER_AXES[i]] THEN
        AllAxesHomed := FALSE;
        EXIT;
    END_IF
END_FOR

// Clear Aborted flag when a new homing command is issued
IF homeSingleRising OR homeAllRising THEN
    Aborted := FALSE;
END_IF

// Abort: halt all active moves immediately when requested while busy.
IF Abort AND Busy AND NOT haltActive THEN
    fbMoveSingle(Execute := FALSE, AxisRef := selectedAxisRef);
    fbMoveNonKin(Execute := FALSE);
    // Clear any pending kin home request
    GVL_GCode.bHomeRequest := FALSE;
    state        := 0;
    Done         := FALSE;
    haltActive   := TRUE;
    FOR i := 1 TO GVL_Axes.NON_KIN_MASTER_COUNT DO
        haltDone[i] := FALSE;
    END_FOR
END_IF

// ============================================================================
// STATE MACHINE
// ============================================================================
CASE state OF

    // ─────────────────────────────────────────────────────────────────────
    0: // IDLE
        Done := FALSE;
        Busy := FALSE;
        Error := FALSE;
        ErrorAxisID := 0;

        IF NOT Enable THEN
            ;   // Stay idle when disabled

        ELSIF homeSingleRising AND HomeAxis > 0 AND HomeAxis <= 36 THEN
            IF NOT isMaster THEN
                InvalidAxis := TRUE;
            ELSIF selectedAxisIsKin THEN
                // Kin axis — dispatch home scripts via trigger_dispatch.py
                // This homes ALL kin axes (Wire1-4 and Wire5_6) since individual
                // kin axis moves are not possible while the group is enabled.
                selectedAxis := HomeAxis;
                state := 20;
                Busy := TRUE;
            ELSE
                // Non-kin axis — direct MC_MoveAbsolute
                selectedAxis := HomeAxis;
                state := 1;
                Busy := TRUE;
            END_IF

        ELSIF homeAllRising THEN
            state := 10;
            Busy := TRUE;
        END_IF

    // =========================================================================
    // SINGLE NON-KIN AXIS HOMING (states 1-2)
    // =========================================================================
    1: // Execute MC_MoveAbsolute to position 0
        fbMoveSingle(
            Execute := TRUE,
            AxisRef := selectedAxisRef,
            Position := 0.0,
            Velocity := Velocity,
            Acceleration := Acceleration,
            Deceleration := Deceleration,
            Jerk := Jerk
        );

        IF fbMoveSingle.Done THEN
            state := 2;
            Done := TRUE;
            Busy := FALSE;
        ELSIF fbMoveSingle.Error THEN
            FC_ReportFault('FB_AxisHome', selectedAxis, fbMoveSingle.ErrorID,
                           'Single axis home move failed', E_MotionStep.WAIT_FOR_LOAD, 0);
            state := 99;
            Error := TRUE;
            ErrorAxisID := selectedAxis;
            Busy := FALSE;
        END_IF

    2: // Wait for command to clear
        fbMoveSingle(Execute := FALSE, AxisRef := selectedAxisRef);
        IF NOT HomeSingle THEN
            state := 0;
            Done := FALSE;
        END_IF

    // =========================================================================
    // HOME ALL AXES (states 10-15)
    // Step 1: Non-kin masters in parallel (MC_MoveAbsolute).
    // Step 2: All kin masters via GVL_GCode.bHomeRequest → trigger_dispatch.py.
    // =========================================================================
    10: // Home 6 non-kin masters simultaneously
        fbMoveNonKin(
            Execute := TRUE,
            Velocity := Velocity,
            Acceleration := Acceleration,
            Deceleration := Deceleration,
            Jerk := Jerk,
            Enable1 := TRUE, Axis1 := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[1]], Position1 := 0.0,
            Enable2 := TRUE, Axis2 := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[2]], Position2 := 0.0,
            Enable3 := TRUE, Axis3 := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[3]], Position3 := 0.0,
            Enable4 := TRUE, Axis4 := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[4]], Position4 := 0.0,
            Enable5 := TRUE, Axis5 := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[5]], Position5 := 0.0,
            Enable6 := TRUE, Axis6 := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[6]], Position6 := 0.0,
            Enable7 := FALSE,
            Enable8 := FALSE
        );

        IF fbMoveNonKin.Done THEN
            state := 11;
        ELSIF fbMoveNonKin.Error THEN
            ErrorAxisID := GVL_Axes.NON_KIN_MASTER_AXES[fbMoveNonKin.ErrorAxisNum];
            FC_ReportFault('FB_AxisHome', ErrorAxisID, fbMoveNonKin.ErrorID,
                           'Home all: non-kin group move failed', E_MotionStep.WAIT_FOR_LOAD, 0);
            state := 99;
            Error := TRUE;
            Busy := FALSE;
        END_IF

    11: // Non-kin done — request kin home via trigger_dispatch.py
        fbMoveNonKin(Execute := FALSE);
        GVL_GCode.bHomeDone    := FALSE;
        GVL_GCode.bHomeRequest := TRUE;
        state := 12;

    12: // Wait for trigger_dispatch.py to home all kin axes and set bHomeDone
        IF GVL_GCode.bHomeDone THEN
            GVL_GCode.bHomeRequest := FALSE;
            GVL_GCode.bHomeDone    := FALSE;
            state := 13;
        END_IF
        // trigger_dispatch.py clears bHomeRequest itself after setting bHomeDone,
        // but we clear it here too as a belt-and-suspenders guard.

    13: // All axes homed
        Done := TRUE;
        Busy := FALSE;
        state := 14;

    14: // Wait for command to clear
        IF NOT HomeAll THEN
            state := 0;
            Done := FALSE;
        END_IF

    // =========================================================================
    // SINGLE KIN AXIS HOMING (states 20-22)
    // Cannot home individual kin axes — homes all kin groups via script dispatch.
    // =========================================================================
    20: // Request kin home via trigger_dispatch.py
        GVL_GCode.bHomeDone    := FALSE;
        GVL_GCode.bHomeRequest := TRUE;
        state := 21;

    21: // Wait for bHomeDone
        IF GVL_GCode.bHomeDone THEN
            GVL_GCode.bHomeRequest := FALSE;
            GVL_GCode.bHomeDone    := FALSE;
            state := 22;
            Done := TRUE;
            Busy := FALSE;
        END_IF

    22: // Wait for command to clear
        IF NOT HomeSingle THEN
            state := 0;
            Done := FALSE;
        END_IF

    // =========================================================================
    // ERROR STATE
    // =========================================================================
    99:
        fbMoveSingle(Execute := FALSE, AxisRef := selectedAxisRef);
        fbMoveNonKin(Execute := FALSE);
        GVL_GCode.bHomeRequest := FALSE;

        IF NOT HomeSingle AND NOT HomeAll THEN
            state := 0;
            Error := FALSE;
            ErrorAxisID := 0;
        END_IF

END_CASE

// ============================================================================
// HALT SEQUENCE
// Runs every scan until all non-kin axes stop. Kin axes are halted by
// ML_KinInterrupt2 called from FB_KinematicEnable (E-stop path).
// ============================================================================
IF haltActive THEN
    FOR i := 1 TO GVL_Axes.NON_KIN_MASTER_COUNT DO
        IF NOT haltDone[i] THEN
            fbHalt[i](
                Execute      := TRUE,
                Axis         := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[i]],
                Deceleration := Deceleration,
                Jerk         := Jerk
            );
            IF fbHalt[i].Done OR fbHalt[i].Error THEN
                haltDone[i] := TRUE;
                fbHalt[i](Execute := FALSE, Axis := AXIF_CONFIG_INDEXES[GVL_Axes.NON_KIN_MASTER_AXES[i]]);
            END_IF
        END_IF
    END_FOR

    haltAllDone := TRUE;
    FOR i := 1 TO GVL_Axes.NON_KIN_MASTER_COUNT DO
        IF NOT haltDone[i] THEN
            haltAllDone := FALSE;
            EXIT;
        END_IF
    END_FOR

    IF haltAllDone THEN
        haltActive := FALSE;
        Busy       := FALSE;
        Aborted    := TRUE;
        FOR i := 1 TO GVL_Axes.NON_KIN_MASTER_COUNT DO
            haltDone[i] := FALSE;
        END_FOR
    END_IF
END_IF

// Reset when disabled (unless halt is still running)
IF NOT Enable AND NOT haltActive THEN
    fbMoveSingle(Execute := FALSE, AxisRef := selectedAxisRef);
    fbMoveNonKin(Execute := FALSE);
    GVL_GCode.bHomeRequest := FALSE;
    state := 0;
    Busy := FALSE;
    Done := FALSE;
    Error := FALSE;
END_IF
