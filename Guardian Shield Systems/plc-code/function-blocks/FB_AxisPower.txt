{attribute 'no_assign'}
FUNCTION_BLOCK FB_AxisPower
// Centralized power management for all 36 axes
// Handles MC_Power enable and MC_Reset for fault clearing
// Uses CXA_Datalayer DL_WriteNode to clear all errors on the Core

VAR_INPUT
    PowerOn : BOOL;         // Enable power to all drives
    Reset : BOOL;           // Reset drive faults
END_VAR

VAR_OUTPUT
    DrivesReady : BOOL;     // TRUE when all drives are powered and ready
    AllResetDone : BOOL;    // TRUE when all resets complete
    ErrorAxisID : UINT;     // First axis with error (0 = none)
END_VAR

VAR
    fbPower : ARRAY[MOTIF_CONFIG.MIN_AXIS_INDEX..MOTIF_CONFIG.MAX_AXIS_INDEX] OF MC_Power;
    fbReset : ARRAY[MOTIF_CONFIG.MIN_AXIS_INDEX..MOTIF_CONFIG.MAX_AXIS_INDEX] OF MC_Reset;

    // CXA_Datalayer for global error reset
    fbPLC_Reset : CXA_Datalayer.DL_WriteNode;

    i : UINT;
    allReady : BOOL;
    allReset : BOOL;
    firstError : UINT;
    resetState : UINT := 0;          // 0=idle, 1=clearing errors, 2=axis reset, 3=wait for axis reset
    rtrigReset : R_TRIG;            // Rising edge detection for Reset input
END_VAR

///////////////////////////////////////////////////////////////////

// Detect rising edge of Reset
rtrigReset(CLK := Reset);

// Reset state machine
IF Reset THEN
    CASE resetState OF
        0: // Idle - start reset sequence on rising edge
            IF rtrigReset.Q THEN
                AllResetDone := FALSE;
                resetState := 1;
            END_IF

        1: // Clear all errors via Datalayer
            fbPLC_Reset(
                Execute := TRUE,
                NodeName := 'diagnosis/confirm/all-errors'
            );

            IF fbPLC_Reset.Done OR fbPLC_Reset.Error THEN
                fbPLC_Reset(Execute := FALSE);
                resetState := 2;
            END_IF

        2: // Axis resets via MC_Reset - execute all at once
            FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
                fbReset[i](
                    Execute := TRUE,
                    Axis := AXIF_CONFIG_INDEXES[i]
                );
            END_FOR
            resetState := 3;

        3: // Wait for all axis resets to complete
            // Keep calling MC_Reset with Execute TRUE
            FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
                fbReset[i](
                    Execute := TRUE,
                    Axis := AXIF_CONFIG_INDEXES[i]
                );
            END_FOR

            // Check if all resets are done
            allReset := TRUE;
            FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
                IF NOT fbReset[i].Done THEN
                    allReset := FALSE;
                    EXIT;
                END_IF
            END_FOR

            IF allReset THEN
                // All resets complete - clear Execute and finish
                FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
                    fbReset[i](
                        Execute := FALSE,
                        Axis := AXIF_CONFIG_INDEXES[i]
                    );
                END_FOR
                AllResetDone := TRUE;
                resetState := 0;
            END_IF
    END_CASE
ELSE
    // Reset not active - clear state and MC_Reset execute flags
    resetState := 0;
    fbPLC_Reset(Execute := FALSE);
    FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
        fbReset[i](
            Execute := FALSE,
            Axis := AXIF_CONFIG_INDEXES[i]
        );
    END_FOR
END_IF

// Power all axes
FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
    fbPower[i](
        Enable := PowerOn,
        Axis := AXIF_CONFIG_INDEXES[i]
    );
END_FOR

// Check if all drives are ready
allReady := TRUE;
allReset := TRUE;
firstError := 0;

FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
    // Check power status
    IF PowerOn AND NOT fbPower[i].Status THEN
        allReady := FALSE;
    END_IF

    // Check for errors
    IF fbPower[i].Error AND firstError = 0 THEN
        firstError := i;
    END_IF

    // Check reset status
    IF Reset AND NOT fbReset[i].Done THEN
        allReset := FALSE;
    END_IF
END_FOR

// Update outputs
DrivesReady := PowerOn AND allReady AND (firstError = 0);
ErrorAxisID := firstError;
