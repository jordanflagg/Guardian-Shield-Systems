FUNCTION_BLOCK FB_ButtonManager
// Button input processing with debouncing and edge detection
// Outputs STAY LATCHED until another button is pressed (no auto-clear)
// This allows state machines to complete their full sequences

VAR_INPUT
    btn_Start_In : BOOL;
    btn_Stop_In : BOOL;
    btn_Home_In : BOOL;
    btn_Manual_In : BOOL;
    btn_Reset_In : BOOL;
    btn_EStop_In : BOOL;
    ClearCommand : BOOL;                // External clear (from state machine when command processed)
END_VAR

VAR_OUTPUT
    btn_Start_Out : BOOL;
    btn_Stop_Out : BOOL;
    btn_Home_Out : BOOL;
    btn_Manual_Out : BOOL;
    btn_Reset_Out : BOOL;
    btn_EStop_Out : BOOL;
    activeCommand : E_ActiveCommand;
END_VAR

VAR
    trig_Start : R_TRIG;
    trig_Stop : R_TRIG;
    trig_Home : R_TRIG;
    trig_Manual : R_TRIG;
    trig_Reset : R_TRIG;
END_VAR


/////////////////////////////////////////////////////////////////

// ============================================================================
// Edge detection
// ============================================================================
trig_Start(CLK := btn_Start_In);
trig_Stop(CLK := btn_Stop_In);
trig_Home(CLK := btn_Home_In);
trig_Manual(CLK := btn_Manual_In);
trig_Reset(CLK := btn_Reset_In);

// ============================================================================
// Clear command only if explicitly requested
// ============================================================================
IF ClearCommand THEN
    activeCommand := E_ActiveCommand.CMD_NONE;
END_IF

// ============================================================================
// Determine active command based on button presses
// Priority: E-Stop > Stop > Start > Home > Manual > Reset
// Command stays latched until another button is pressed
// ============================================================================
IF btn_EStop_In THEN
    // E-stop overrides everything
    activeCommand := E_ActiveCommand.CMD_NONE;

ELSIF trig_Stop.Q THEN
    // Stop has priority over start
    activeCommand := E_ActiveCommand.CMD_STOP;

ELSIF trig_Start.Q THEN
    activeCommand := E_ActiveCommand.CMD_START;

ELSIF trig_Home.Q THEN
    activeCommand := E_ActiveCommand.CMD_HOME;

ELSIF trig_Manual.Q THEN
    activeCommand := E_ActiveCommand.CMD_MANUAL;

ELSIF trig_Reset.Q THEN
    activeCommand := E_ActiveCommand.CMD_RESET;

END_IF

// ============================================================================
// EXPLICIT: Set outputs based on active command (clear all, set one)
// ============================================================================
CASE activeCommand OF
    
    E_ActiveCommand.CMD_START:
        btn_Start_Out := TRUE;
        btn_Stop_Out := FALSE;   
        btn_Home_Out := FALSE;   
        btn_Manual_Out := FALSE; 
        btn_Reset_Out := FALSE;  // ← Explicitly FALSE
    
    E_ActiveCommand.CMD_STOP:
        btn_Start_Out := FALSE;
        btn_Stop_Out := TRUE;
        btn_Home_Out := FALSE;
        btn_Manual_Out := FALSE;
        btn_Reset_Out := FALSE;
    
    E_ActiveCommand.CMD_HOME:
        btn_Start_Out := FALSE;
        btn_Stop_Out := FALSE;
        btn_Home_Out := TRUE;
        btn_Manual_Out := FALSE;
        btn_Reset_Out := FALSE;
    
    E_ActiveCommand.CMD_MANUAL:
        btn_Start_Out := FALSE;
        btn_Stop_Out := FALSE;
        btn_Home_Out := FALSE;
        btn_Manual_Out := TRUE;
        btn_Reset_Out := FALSE;
    
    E_ActiveCommand.CMD_RESET:
        btn_Start_Out  := FALSE;
        btn_Stop_Out   := FALSE;
        btn_Home_Out   := FALSE;
        btn_Manual_Out := FALSE;
        btn_Reset_Out  := TRUE;
        activeCommand  := E_ActiveCommand.CMD_NONE;  // Self-clearing: one scan pulse only
    
    E_ActiveCommand.CMD_NONE:
        btn_Start_Out := FALSE;
        btn_Stop_Out := FALSE;
        btn_Home_Out := FALSE;
        btn_Manual_Out := FALSE;
        btn_Reset_Out := FALSE;

END_CASE

// E-stop passes through directly (level-sensitive, not edge)
btn_EStop_Out := btn_EStop_In;
