{attribute 'no_assign'}
FUNCTION_BLOCK FB_KinCommand
(*
==============================================================================
  FB_KinCommand
  Sends a parameterless command to a ctrlX kinematic group via DL_WriteNode.

  Constructs the Data Layer path:
    motion/kin/{KinName}/cmd/{Command}

  Used for:
    Command = 'reset'    — clear ERRORSTOP state
    Command = 'group-en' — enable kinematic group (required after cold start
                           or after reset before G-code can run)

  Edge-triggered: Execute rising edge fires the call once.
  Reset Execute to FALSE after Done/Error before re-using.
==============================================================================
*)

VAR_INPUT
    Execute : BOOL;
    KinName : STRING(20);       // e.g. 'Wire1', 'Wire2'
    Command : STRING(20);       // e.g. 'reset', 'group-en'
END_VAR

VAR_OUTPUT
    Done  : BOOL;
    Busy  : BOOL;
    Error : BOOL;
END_VAR

VAR
    fbCall   : DL_WriteNode;
    nodePath : STRING(200);
    dummy    : BYTE := 0;       // Passed to DL_WriteNode for parameterless commands (ANY param)
END_VAR

///////////////////////////////////////////////////////////////////////////////

nodePath := CONCAT('motion/kin/', CONCAT(KinName, CONCAT('/cmd/', Command)));


fbCall(
    Execute  := Execute,
    NodeName := nodePath,
    Value   := dummy
);

Done  := fbCall.Done;
Busy  := fbCall.Active;
Error := fbCall.Error;
