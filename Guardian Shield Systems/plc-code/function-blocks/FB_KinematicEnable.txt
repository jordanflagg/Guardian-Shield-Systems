{attribute 'no_assign'}
FUNCTION_BLOCK FB_KinematicEnable
(*
==============================================================================
  FB_KinematicEnable  v3.0 — Always-Enabled Kinematics
  Called from PLC_PRG every scan.

  Wire1-4 and Wire5_6 kinematic groups are enabled once when drives first
  come ready (rising edge of DrivesReady) and remain enabled for the entire
  machine session. They are never disabled between cutting groups — with
  Wire1-4 and Wire5_6 having zero axis overlap there is no reason to.

  TWO SECTIONS:

  1. KIN ENABLE (rising edge of DrivesReady)
       Calls ML_KinEnable for Wire1, Wire2, Wire3, Wire4, Wire5_6 on the
       first scan that DrivesReady is TRUE. If drives go off and come back,
       the rising edge fires again and re-enables all kins automatically.

  2. KIN OPSTATE MONITOR (24/7, always runs)
       Polls motion/kin/WireN/state/opstate/plcopen for all 5 kins every
       ~2 seconds via DL_ReadNode. Reports ERRORSTOP via FC_ReportFault.
       Self-clears when kin leaves ERRORSTOP.

  INTERRUPT / CONTINUE (only during active cutting — GVL_GCode.bBusy = TRUE):
    InterruptKinematics — ML_KinInterrupt2 on all active kins.
                          Called by E-stop or Stop during cutting step.
                          When drives come back (DrivesReady rises), kins
                          are re-enabled by section 1. ML_KinContinue is NOT
                          called automatically — it fires only when the Start
                          button is pressed again (FB_MachineStateMachine
                          pulses ContinueKinematics when resuming).
    ContinueKinematics  — ML_KinContinue on all kins. Only fires when
                          GVL_GCode.bBusy = TRUE (active cutting).
==============================================================================
*)

VAR_INPUT
    DrivesReady          : BOOL;   // From FB_AxisPower — enable kins when drives are ready
    InterruptKinematics  : BOOL;   // Hold TRUE to interrupt kins mid-cut (E-stop / Stop)
    ContinueKinematics   : BOOL;   // Pulse TRUE to resume interrupted kins (Start press)
    DbgKinEnableNum      : UINT;   // 0=no-op, 1-5: enable that wire's kin (debug/test use)
    DbgKinDisableNum     : UINT;   // 0=no-op, 1-5: disable that wire's kin (debug/test use)
END_VAR

// No outputs needed — kins are always on; opstate monitor reports faults directly

VAR
    // ─── Kin enable on DrivesReady rising edge ────────────────────────────
    drivesReadyPrev    : BOOL;

    fbKinEn      : ARRAY[1..5] OF ML_KinEnableData;
    fbKinDis     : ARRAY[1..5] OF ML_KinDisableData;
    fbKinInt     : ARRAY[1..5] OF ML_KinInterruptData2;
    fbKinCont    : ARRAY[1..5] OF ML_KinContinueData;
    namesSet     : BOOL := FALSE;

    // ─── 24/7 Opstate Monitor ────────────────────────────────────────────────
    monState     : UINT := 0;
    monTimer     : TON;
    fbOp         : ARRAY[1..5] OF CXA_Datalayer.DL_ReadNode;
    sOpstate     : ARRAY[1..5] OF STRING(20);
    kinErrRep    : ARRAY[1..5] OF BOOL;   // latched per-kin; clears when ERRORSTOP goes away
    i            : UINT;

    sNodeOp      : ARRAY[1..5] OF STRING(80);
    nodeNamesSet : BOOL := FALSE;
END_VAR

///////////////////////////////////////////////////////////////////////////////

// ============================================================================
// FILL NAMES (once)
// ============================================================================
IF NOT namesSet THEN
    fbKinEn[1].In.KinName := 'Wire1';    fbKinEn[1].In.Source := 'FB_KinematicEnable';
    fbKinEn[2].In.KinName := 'Wire2';    fbKinEn[2].In.Source := 'FB_KinematicEnable';
    fbKinEn[3].In.KinName := 'Wire3';    fbKinEn[3].In.Source := 'FB_KinematicEnable';
    fbKinEn[4].In.KinName := 'Wire4';    fbKinEn[4].In.Source := 'FB_KinematicEnable';
    fbKinEn[5].In.KinName := 'Wire5_6';  fbKinEn[5].In.Source := 'FB_KinematicEnable';

    fbKinDis[1].In.KinName := 'Wire1';    fbKinDis[1].In.Source := 'FB_KinematicEnable';
    fbKinDis[2].In.KinName := 'Wire2';    fbKinDis[2].In.Source := 'FB_KinematicEnable';
    fbKinDis[3].In.KinName := 'Wire3';    fbKinDis[3].In.Source := 'FB_KinematicEnable';
    fbKinDis[4].In.KinName := 'Wire4';    fbKinDis[4].In.Source := 'FB_KinematicEnable';
    fbKinDis[5].In.KinName := 'Wire5_6';  fbKinDis[5].In.Source := 'FB_KinematicEnable';

    fbKinInt[1].In.KinName := 'Wire1';    fbKinInt[1].In.Source := 'FB_KinematicEnable';
    fbKinInt[2].In.KinName := 'Wire2';    fbKinInt[2].In.Source := 'FB_KinematicEnable';
    fbKinInt[3].In.KinName := 'Wire3';    fbKinInt[3].In.Source := 'FB_KinematicEnable';
    fbKinInt[4].In.KinName := 'Wire4';    fbKinInt[4].In.Source := 'FB_KinematicEnable';
    fbKinInt[5].In.KinName := 'Wire5_6';  fbKinInt[5].In.Source := 'FB_KinematicEnable';

    fbKinCont[1].In.KinName := 'Wire1';    fbKinCont[1].In.Source := 'FB_KinematicEnable';
    fbKinCont[2].In.KinName := 'Wire2';    fbKinCont[2].In.Source := 'FB_KinematicEnable';
    fbKinCont[3].In.KinName := 'Wire3';    fbKinCont[3].In.Source := 'FB_KinematicEnable';
    fbKinCont[4].In.KinName := 'Wire4';    fbKinCont[4].In.Source := 'FB_KinematicEnable';
    fbKinCont[5].In.KinName := 'Wire5_6';  fbKinCont[5].In.Source := 'FB_KinematicEnable';

    namesSet := TRUE;
END_IF

IF NOT nodeNamesSet THEN
    sNodeOp[1] := 'motion/kin/Wire1/state/opstate/plcopen';
    sNodeOp[2] := 'motion/kin/Wire2/state/opstate/plcopen';
    sNodeOp[3] := 'motion/kin/Wire3/state/opstate/plcopen';
    sNodeOp[4] := 'motion/kin/Wire4/state/opstate/plcopen';
    sNodeOp[5] := 'motion/kin/Wire5_6/state/opstate/plcopen';
    nodeNamesSet := TRUE;
END_IF

// ============================================================================
// SECTION 1 — KIN ENABLE ON DRIVES READY
// Enable all 5 kins on the rising edge of DrivesReady (drives first come up,
// or drives cycle off and back on after E-stop/fault recovery).
// ============================================================================
IF DrivesReady AND NOT drivesReadyPrev THEN
    CXA_MOTION.ML_KinEnable(Data := fbKinEn[1]);   // Wire1
    CXA_MOTION.ML_KinEnable(Data := fbKinEn[2]);   // Wire2
    CXA_MOTION.ML_KinEnable(Data := fbKinEn[3]);   // Wire3
    CXA_MOTION.ML_KinEnable(Data := fbKinEn[4]);   // Wire4
    CXA_MOTION.ML_KinEnable(Data := fbKinEn[5]);   // Wire5_6
END_IF
drivesReadyPrev := DrivesReady;

// ============================================================================
// SECTION 2 — INTERRUPT / CONTINUE
// Only fires during active cutting (bBusy = TRUE).
// ML_KinInterrupt2 is only valid in MOVING state — the bBusy guard prevents
// calling it on STANDBY kins (which would cause "Invalid state" logbook errors).
// ML_KinContinue is gated the same way: no-op if kins are not interrupted.
// ============================================================================
IF GVL_GCode.bBusy THEN
    IF InterruptKinematics THEN
        FOR i := 1 TO 5 DO
            CXA_MOTION.ML_KinInterrupt2(Data := fbKinInt[i]);
        END_FOR
    END_IF

    IF ContinueKinematics THEN
        FOR i := 1 TO 5 DO
            CXA_MOTION.ML_KinContinue(Data := fbKinCont[i]);
        END_FOR
    END_IF
END_IF

// ============================================================================
// SECTION 3 — DEBUG ENABLE / DISABLE (HMI Kin Test tab)
// HMI pulses DbgKinEnableNum / DbgKinDisableNum with a wire number (1-5)
// for ~500 ms; we call ML_KinEnable / ML_KinDisable each scan it is set.
// ============================================================================
IF DbgKinEnableNum >= 1 AND DbgKinEnableNum <= 5 THEN
    CXA_MOTION.ML_KinEnable(Data := fbKinEn[DbgKinEnableNum]);
END_IF
IF DbgKinDisableNum >= 1 AND DbgKinDisableNum <= 5 THEN
    CXA_MOTION.ML_KinDisable(Data := fbKinDis[DbgKinDisableNum]);
END_IF

// ============================================================================
// SECTION 4 — KIN OPSTATE MONITOR (24/7, always runs)
// Polls all 5 kin PLCopen states every ~2 seconds.
// Reports ERRORSTOP via FC_ReportFault; self-clears on recovery.
// ============================================================================
CASE monState OF

    0: // Kick off all 5 reads simultaneously
        FOR i := 1 TO 5 DO
            fbOp[i](Execute := TRUE, NodeName := sNodeOp[i], Value := sOpstate[i]);
        END_FOR
        monState := 1;

    1: // Wait for all reads to finish (Done or Error)
        FOR i := 1 TO 5 DO
            fbOp[i](Execute := TRUE, NodeName := sNodeOp[i], Value := sOpstate[i]);
        END_FOR
        IF fbOp[1].Done OR fbOp[1].Error THEN
        IF fbOp[2].Done OR fbOp[2].Error THEN
        IF fbOp[3].Done OR fbOp[3].Error THEN
        IF fbOp[4].Done OR fbOp[4].Error THEN
        IF fbOp[5].Done OR fbOp[5].Error THEN
            FOR i := 1 TO 5 DO
                fbOp[i](Execute := FALSE, NodeName := sNodeOp[i], Value := sOpstate[i]);
            END_FOR
            monState := 2;
        END_IF END_IF END_IF END_IF END_IF

    2: // Check results — report ERRORSTOP, self-clear on recovery
        FOR i := 1 TO 5 DO
            IF sOpstate[i] <> 'ERRORSTOP' THEN
                kinErrRep[i] := FALSE;   // Recovered — allow re-reporting if it faults again
            ELSIF NOT kinErrRep[i] THEN
                FC_ReportFault(
                    source        := 'FB_KinematicEnable',
                    axisID        := 0,
                    errorCode     := DWORD#0,
                    description   := CONCAT(sNodeOp[i], ' is in ERRORSTOP'),
                    motionStep    := E_MotionStep.WAIT_FOR_LOAD,
                    motionSubStep := 0
                );
                kinErrRep[i] := TRUE;
            END_IF
        END_FOR
        monTimer(IN := FALSE);   // Reset timer
        monState := 3;

    3: // Wait ~2s before next poll cycle
        monTimer(IN := TRUE, PT := T#2s);
        IF monTimer.Q THEN
            monTimer(IN := FALSE);
            monState := 0;
        END_IF

END_CASE
