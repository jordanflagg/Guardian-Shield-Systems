{attribute 'no_assign'}
FUNCTION_BLOCK FB_MachineStateMachine

VAR_INPUT
    startButton : BOOL;
    stopButton : BOOL;
    homeButton : BOOL;
    manualMode : BOOL;
    eStopActive : BOOL;
    resetButton : BOOL;
    DrivesReady : BOOL;
    BlockLoaded : BOOL;
    FilesReady : BOOL;
    HasFault : BOOL;    // From FB_FaultMonitor — any fault in queue (power, motion, or ctrlX)
END_VAR

VAR_OUTPUT
    currentStateOut : E_MachineState;   // So HMI and other code can see it
    isRunning : BOOL;
    isFaulted : BOOL;
    isManualMode : BOOL;
    isHomed : BOOL;

    // Motion sequence visualization (pass-through from FB_MotionSequence)
    motionStepOut : E_MotionStep;       // Current motion step
    motionSubStep : UINT;               // Current substep
    activeAxesMask : LWORD;             // Bitmask of moving axes
    stepDescription : STRING[80];       // Human-readable step description
    activeMotionStepOut : E_MotionStep; // Only valid while the motion sequence is actively executing
    activeMotionSubStep : UINT;         // Only valid while the motion sequence is actively executing
    resumeMotionStepOut : E_MotionStep; // Last resumable step when the cycle is paused
    resumeMotionSubStep : UINT;         // Last resumable substep when the cycle is paused

    // Kinematic interrupt/continue — consumed by FB_KinematicEnable in PLC_PRG
    interruptKinematics : BOOL;         // TRUE while wire kins should be held interrupted
    continueKinematics  : BOOL;         // Pulse TRUE for one scan to resume interrupted kins

    // Cycle paused state — TRUE when Stop was pressed mid-cycle
    cyclePausedOut : BOOL;

    // Public jog feedback for HMI
    jogActiveOut : BOOL;
    jogErrorOut : BOOL;
    jogInvalidAxisOut : BOOL;

    // Public homing feedback for HMI
    homeBusyOut : BOOL;
    homeDoneOut : BOOL;
    homeErrorOut : BOOL;
    homeErrorAxisIDOut : UINT;
    homeInvalidAxisOut : BOOL;
    allAxesHomedOut : BOOL;
    axisIsHomedOut : ARRAY[1..36] OF BOOL;
END_VAR

VAR
    currentState : E_MachineState := E_MachineState.INIT;

    motionSequence : FB_MotionSequence;
    axisManual : FB_AxisManual;
    fbHome : FB_AxisHome;

    // Emergency halt function blocks
    fbHalt : ARRAY[MOTIF_CONFIG.MIN_AXIS_INDEX..MOTIF_CONFIG.MAX_AXIS_INDEX] OF MC_Halt;
    i : UINT;

    // Resume tracking
    kinematicsPaused : BOOL;
    cyclePaused : BOOL;
    resumeStep : E_MotionStep := E_MotionStep.WAIT_FOR_LOAD;
    resumeSubStep : UINT;
    resumeDescription : STRING[80];
END_VAR

/////////////////////////////////////////////////////////////////////

// Default outputs every scan; active states set what they need explicitly.
isRunning := FALSE;
isFaulted := FALSE;
isManualMode := FALSE;
activeMotionStepOut := E_MotionStep.WAIT_FOR_LOAD;
activeMotionSubStep := 0;
activeAxesMask := 0;
interruptKinematics := FALSE;
continueKinematics := FALSE;

// Check for E-stop - can interrupt any state except E_STOPPED itself
IF eStopActive AND (currentState <> E_MachineState.E_STOPPED) THEN
    IF currentState = E_MachineState.RUNNING AND
       motionSequence.CurrentStep = E_MotionStep.CUTTING THEN
        interruptKinematics := TRUE;
        GVL_GCode.bPauseRequested := FALSE;
        GVL_GCode.bAbortRequested := TRUE;
        GVL_GCode.bBusy := FALSE;
        kinematicsPaused := FALSE;
        motionSequence(Enable := FALSE, ResetSequence := FALSE, BlockLoaded := BlockLoaded, FilesReady := FilesReady);
    END_IF
    currentState := E_MachineState.E_STOPPED;
END_IF;

// Check for fault — interrupts any state except FAULTED/E_STOPPED/RESETTING.
// E-stop takes priority over fault (handled above), so this only fires when
// eStopActive is FALSE. RESETTING is excluded so the reset sequence can clear
// the queue before HasFault goes FALSE.
IF HasFault
   AND (currentState <> E_MachineState.FAULTED)
   AND (currentState <> E_MachineState.E_STOPPED)
   AND (currentState <> E_MachineState.RESETTING) THEN
    IF currentState = E_MachineState.RUNNING AND
       motionSequence.CurrentStep = E_MotionStep.CUTTING THEN
        interruptKinematics := TRUE;
        GVL_GCode.bPauseRequested := FALSE;
        GVL_GCode.bAbortRequested := TRUE;
        GVL_GCode.bBusy     := FALSE;
    END_IF
    currentState := E_MachineState.FAULTED;
END_IF;

// Main state machine
CASE currentState OF

    // ------------------------------------------------------------------------
    E_MachineState.INIT:
        isRunning := FALSE;
        isFaulted := FALSE;
        currentState := E_MachineState.IDLE;

    // ------------------------------------------------------------------------
    E_MachineState.IDLE:
        IF homeButton THEN
            IF DrivesReady THEN
                currentState := E_MachineState.HOMING;
            END_IF

        ELSIF manualMode THEN
            IF DrivesReady THEN
                currentState := E_MachineState.MANUAL;
            END_IF

        ELSIF startButton THEN
            IF NOT DrivesReady THEN
                GVL_GCode.sStatus := 'Not ready - enable drives first';
            ELSIF kinematicsPaused THEN
                continueKinematics  := TRUE;
                GVL_GCode.bPauseRequested := FALSE;
                GVL_GCode.bAbortRequested := FALSE;
                kinematicsPaused    := FALSE;
                cyclePaused         := FALSE;
                currentState := E_MachineState.RUNNING;
            ELSIF cyclePaused THEN
                GVL_GCode.bPauseRequested := FALSE;
                GVL_GCode.bAbortRequested := FALSE;
                cyclePaused := FALSE;
                currentState := E_MachineState.RUNNING;
            ELSIF FilesReady THEN
                GVL_GCode.bPauseRequested := FALSE;
                GVL_GCode.bAbortRequested := FALSE;
                currentState := E_MachineState.RUNNING;
            ELSE
                GVL_GCode.sStatus := 'Not ready - upload G-code via pipeline first';
            END_IF;

        END_IF;

        motionStepOut := E_MotionStep.WAIT_FOR_LOAD;
        motionSubStep := 0;
        IF cyclePaused OR kinematicsPaused THEN
            motionStepOut := resumeStep;
            motionSubStep := resumeSubStep;
            stepDescription := resumeDescription;
        ELSE
            IF DrivesReady THEN
                stepDescription := 'Idle';
            ELSE
                stepDescription := 'Idle - drives off';
            END_IF
        END_IF;

    // ------------------------------------------------------------------------
    E_MachineState.HOMING:
        fbHome(
            Enable   := TRUE,
            Abort    := stopButton,
            HomeAxis := GVL_HMI.HMI_HomeAxis,
            HomeSingle := GVL_HMI.HMI_HomeSingle,
            HomeAll := GVL_HMI.HMI_HomeAll,
            Velocity := GVL_HMI.HMI_HomeSpeed,
            Acceleration := 5.0,
            Deceleration := 5.0,
            Jerk := 10.0,
            PositionTolerance := 0.5
        );

        isHomed := fbHome.AllAxesHomed;
        motionStepOut := E_MotionStep.WAIT_FOR_LOAD;
        motionSubStep := 0;
        stepDescription := 'Homing';

        IF fbHome.Error THEN
            currentState := E_MachineState.FAULTED;
        END_IF

        IF stopButton THEN
            fbHome(Enable := FALSE, Abort := stopButton);
            currentState := E_MachineState.IDLE;
        END_IF

    // ------------------------------------------------------------------------
    E_MachineState.MANUAL:
        isManualMode := TRUE;

        axisManual(
            Enable := isManualMode,
            AxisSelect := GVL_HMI.HMI_JogAxis,
            JogPositive := GVL_HMI.HMI_JogEnable AND GVL_HMI.HMI_JogDirection,
            JogNegative := GVL_HMI.HMI_JogEnable AND NOT GVL_HMI.HMI_JogDirection,
            JogVelocity := GVL_HMI.HMI_JogSpeed
        );

        motionStepOut := E_MotionStep.WAIT_FOR_LOAD;
        motionSubStep := 0;
        stepDescription := 'Manual jog mode';

        IF manualMode OR stopButton THEN
            currentState := E_MachineState.IDLE;
        END_IF;

    // ------------------------------------------------------------------------
    E_MachineState.RUNNING:
        isRunning := TRUE;
        continueKinematics := FALSE;

        motionSequence(Enable := TRUE, ResetSequence := FALSE, BlockLoaded := BlockLoaded, FilesReady := FilesReady);
        activeMotionStepOut := motionSequence.CurrentStep;
        activeMotionSubStep := motionSequence.CurrentSubStep;
        resumeStep := motionSequence.CurrentStep;
        resumeSubStep := motionSequence.CurrentSubStep;
        resumeDescription := motionSequence.StepDescription;
        resumeMotionStepOut := resumeStep;
        resumeMotionSubStep := resumeSubStep;
        motionStepOut := motionSequence.CurrentStep;
        motionSubStep := motionSequence.CurrentSubStep;
        activeAxesMask := motionSequence.ActiveAxesMask;
        stepDescription := motionSequence.StepDescription;

        IF stopButton THEN
            currentState := E_MachineState.STOPPING;
        END_IF

    // ------------------------------------------------------------------------
    E_MachineState.STOPPING:
        IF motionSequence.CurrentStep = E_MotionStep.CUTTING THEN
            interruptKinematics := TRUE;
            kinematicsPaused    := TRUE;
            GVL_GCode.bPauseRequested := TRUE;
            GVL_GCode.bAbortRequested := FALSE;
        END_IF
        cyclePaused := TRUE;
        currentState := E_MachineState.IDLE;

    // ------------------------------------------------------------------------
    E_MachineState.FAULTED:
        isFaulted := TRUE;
        GVL_GCode.bPauseRequested := FALSE;
        GVL_GCode.bAbortRequested := TRUE;
        motionStepOut := resumeStep;
        motionSubStep := resumeSubStep;
        stepDescription := resumeDescription;

        IF resetButton THEN
            currentState := E_MachineState.RESETTING;
        END_IF;

    // ------------------------------------------------------------------------
    E_MachineState.E_STOPPED:
        isFaulted := TRUE;
        GVL_GCode.bPauseRequested := FALSE;
        GVL_GCode.bAbortRequested := TRUE;
        motionStepOut := resumeStep;
        motionSubStep := resumeSubStep;
        stepDescription := resumeDescription;

        FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
            fbHalt[i](
                Execute := eStopActive,
                Deceleration := 1,
                Jerk := 1,
                Axis := AXIF_CONFIG_INDEXES[i]
            );
        END_FOR

        IF NOT eStopActive AND resetButton THEN
            currentState := E_MachineState.RESETTING;
        END_IF;

    // ------------------------------------------------------------------------
    E_MachineState.RESETTING:
        interruptKinematics := FALSE;
        continueKinematics := FALSE;
        kinematicsPaused    := FALSE;
        cyclePaused         := FALSE;
        GVL_GCode.bPauseRequested := FALSE;
        GVL_GCode.bAbortRequested := TRUE;
        motionStepOut := E_MotionStep.WAIT_FOR_LOAD;
        motionSubStep := 0;
        resumeStep := E_MotionStep.WAIT_FOR_LOAD;
        resumeSubStep := 0;
        resumeDescription := 'Idle';
        stepDescription := 'Resetting faults';

        motionSequence(Enable := FALSE, ResetSequence := TRUE, BlockLoaded := BlockLoaded, FilesReady := FilesReady);
        GVL_GCode.bBusy  := FALSE;
        GVL_GCode.bError := FALSE;
        // Hold RESETTING until FB_FaultMonitor acknowledges the clear.
        // FB_FaultMonitor runs BEFORE this FB each scan, reading GVL_HMI.MACHINE_STATE
        // which is written AFTER this FB runs.  So RESETTING must persist for one full
        // scan so faultMonitor can see it, fire its Reset block, and drive HasFault FALSE.
        IF NOT HasFault THEN
            currentState := E_MachineState.IDLE;
        END_IF

END_CASE;

// ============================================================================
// ALWAYS UPDATE AXIS HOMED STATUS
// ============================================================================
IF currentState <> E_MachineState.HOMING THEN
    fbHome(
        Enable := FALSE,
        HomeAxis := 0,
        HomeSingle := FALSE,
        HomeAll := FALSE,
        Velocity := 10.0,
        Acceleration := 5.0,
        Deceleration := 5.0,
        Jerk := 10.0,
        PositionTolerance := 0.5
    );
    isHomed := fbHome.AllAxesHomed;
END_IF

// ============================================================================
// Ensure jog is disabled outside MANUAL mode.
// ============================================================================
IF currentState <> E_MachineState.MANUAL THEN
    axisManual(
        Enable := FALSE,
        AxisSelect := GVL_HMI.HMI_JogAxis,
        JogPositive := FALSE,
        JogNegative := FALSE,
        JogVelocity := GVL_HMI.HMI_JogSpeed
    );
END_IF

// Copy internal state to outputs
currentStateOut := currentState;
resumeMotionStepOut := resumeStep;
resumeMotionSubStep := resumeSubStep;

// Expose cycle paused state to HMI
cyclePausedOut := cyclePaused;
jogActiveOut := axisManual.Active;
jogErrorOut := axisManual.Error;
jogInvalidAxisOut := axisManual.InvalidAxis;
homeBusyOut := fbHome.Busy;
homeDoneOut := fbHome.Done;
homeErrorOut := fbHome.Error;
homeErrorAxisIDOut := fbHome.ErrorAxisID;
homeInvalidAxisOut := fbHome.InvalidAxis;
allAxesHomedOut := fbHome.AllAxesHomed;
FOR i := 1 TO 36 DO
    axisIsHomedOut[i] := fbHome.AxisIsHomed[i];
END_FOR
