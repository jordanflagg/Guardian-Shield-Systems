{attribute 'no_assign'}
FUNCTION_BLOCK FB_MachineStateMachine

VAR_INPUT
    startButton : BOOL;
    stopButton : BOOL;
    homeButton : BOOL;
    manualMode : BOOL;
    eStopActive : BOOL;
    resetButton : BOOL;
END_VAR

VAR_OUTPUT
    currentStateOut : E_MachineState;   // So HMI and other code can see it
    isRunning : BOOL;
    isFaulted : BOOL;
	isManualMode : BOOL;
    isHomed : BOOL;

    // Motion sequence visualization (pass-through from FB_MotionSequence)
    motionStepOut : E_MotionStep;       // Current motion step
    motionSubStep : UNIT;                // Current substep
    activeAxesMask : LWORD;             // Bitmask of moving axes
    stepDescription : STRING[80];       // Human-readable step description

    // Centralized error information (from FB_ErrorHandler)
    hasError : BOOL;                    // Any error occurred (power, gearing, motion)
    errorInfo : ST_MotionError;         // Detailed error information
    errorCount : UNIT;                  // Total error count since last reset

    
END_VAR

VAR
    currentState : E_MachineState := E_MachineState.INIT;  // Internal state

    // Motion sequence controller
    motionSequence : FB_MotionSequence;

    // Centralized error handler (collects errors from all subsystems)
    errorHandler : FB_ErrorHandler;
	
	axisManual : FB_AxisManual;         // Jog control for manual mode
	fbHome : FB_AxisHome;				// Axis control for home mode

    // Emergency halt function blocks
    fbHalt : ARRAY[MOTIF_CONFIG.MIN_AXIS_INDEX..MOTIF_CONFIG.MAX_AXIS_INDEX] OF MC_Halt;
    i : UINT;
END_VAR

/////////////////////////////////////////////////////////////////////


// Check for E-stop - can interrupt any state except E_STOPPED itself
IF eStopActive AND (currentState <> E_MachineState.E_STOPPED) THEN
    currentState := E_MachineState.E_STOPPED;
END_IF;

// Main state machine
CASE currentState OF

    // ------------------------------------------------------------------------
    E_MachineState.INIT:
        // Reset all output flags
        isRunning := FALSE;
        isFaulted := FALSE;
        isHomed := TRUE;  // Assume homed since we have absolute encoders
        
        // Initialize any other variables you need
        
        // Automatically transition to IDLE
        currentState := E_MachineState.IDLE;
    
    // ------------------------------------------------------------------------
    E_MachineState.IDLE:
        // Update outputs
        isRunning := FALSE;

        // Check for operator commands and transition accordingly
        IF homeButton THEN
            currentState := E_MachineState.HOMING;

        ELSIF manualMode THEN
            currentState := E_MachineState.MANUAL;

        ELSIF startButton THEN
            currentState := E_MachineState.RUNNING;

        END_IF;
    
    // ------------------------------------------------------------------------
    E_MachineState.HOMING:
        // Homing mode: move selected axis or all axes to position 0
        // Uses FB_AxisHome with HMI controls

        fbHome(
            Enable := TRUE,
            HomeAxis := PLC_PRG.HMI_HomeAxis,
            HomeSingle := PLC_PRG.HMI_HomeSingle,
            HomeAll := PLC_PRG.HMI_HomeAll,
            Velocity := PLC_PRG.HMI_HomeSpeed,
            Acceleration := 5.0,
            Deceleration := 5.0,
            Jerk := 10.0,
            PositionTolerance := 0.5
        );

        // Update isHomed based on actual axis positions
        isHomed := fbHome.AllAxesHomed;

        // Check for errors during homing
        IF fbHome.Error THEN
            currentState := E_MachineState.FAULTED;
        END_IF

        // Exit homing mode when operator deselects it
        IF NOT homeButton THEN
            fbHome(Enable := FALSE);
            currentState := E_MachineState.IDLE;
        END_IF
    
    // ------------------------------------------------------------------------
    E_MachineState.MANUAL:	
		// ============================================================================
		// HMI MANUAL JOG CONTROL - Only active when machine is in MANUAL state
		// Called directly from PLC_PRG for explicit control
		// ============================================================================
		isManualMode := TRUE;
		
		axisManual(
			Enable := isManualMode,
			AxisSelect := PLC_PRG.HMI_JogAxis,
			JogPositive := PLC_PRG.HMI_JogEnable AND PLC_PRG.HMI_JogDirection,      // Enable + positive direction
			JogNegative := PLC_PRG.HMI_JogEnable AND NOT PLC_PRG.HMI_JogDirection,  // Enable + negative direction
			JogVelocity := PLC_PRG.HMI_JogSpeed
		);
		
        // Exit manual mode when operator deselects it
        IF NOT manualMode THEN
			isManualMode := FALSE;
            currentState := E_MachineState.IDLE;
        END_IF;
    
    // ------------------------------------------------------------------------
    E_MachineState.RUNNING:
        // Update outputs
        isRunning := TRUE;

        // Run motion sequence (gantry groups configured in Bosch web interface)
        motionSequence(
            Enable := TRUE
        );

        // Check if sequence completed
        IF motionSequence.SequenceComplete THEN
            // Continue running for next cycle
        END_IF

        // Check for any errors (motion, gearing, or power) via centralized handler
        IF errorHandler.HasError THEN
            currentState := E_MachineState.FAULTED;
        END_IF

        // Check for stop button
        IF stopButton THEN
            currentState := E_MachineState.STOPPING;
        END_IF
    
    // ------------------------------------------------------------------------
    E_MachineState.STOPPING:
        // Update outputs
        isRunning := FALSE;
		startButton := FALSE;

        // TODO: Command all active axes to decelerate in a controlled manner
        // In real implementation, you'd use MC_Halt for each axis
        // and wait for them to reach zero velocity

        // For now, simplified transition back to IDLE
        currentState := E_MachineState.IDLE;
    
    // ------------------------------------------------------------------------
    E_MachineState.FAULTED:
        // Update outputs
        isRunning := FALSE;
        isFaulted := TRUE;

        // TODO: Log which axis faulted, display error message

        // Wait for operator to acknowledge and reset
        IF resetButton THEN
            currentState := E_MachineState.RESETTING;
        END_IF;
    
    // ------------------------------------------------------------------------
    E_MachineState.E_STOPPED:
        // Update outputs
        isRunning := FALSE;
        isFaulted := TRUE;

        // Immediately halt all axes (emergency stop)
		FOR i := MOTIF_CONFIG.MIN_AXIS_INDEX TO MOTIF_CONFIG.MAX_AXIS_INDEX DO
            fbHalt[i](
                Execute := eStopActive,
				Deceleration := 1,
				Jerk := 1,
                Axis := AXIF_CONFIG_INDEXES[i]
            );
		END_FOR

        // Can only exit when E-stop is released AND operator resets
        IF NOT eStopActive AND resetButton THEN
            currentState := E_MachineState.RESETTING;
        END_IF;
    
    // ------------------------------------------------------------------------
    E_MachineState.RESETTING:
        // Clear fault flags
        isFaulted := FALSE;
        
        // TODO: Reset axis errors, clear alarm buffers, etc.
        
        // Return to IDLE
        currentState := E_MachineState.IDLE;

END_CASE;

// ============================================================================
// ALWAYS UPDATE AXIS HOMED STATUS
// FB_AxisHome reads all axis positions to determine if they're at home position.
// We call it in every scan (even when not in HOMING mode) so the AxisIsHomed
// status is always up-to-date for HMI display. When Enable=FALSE, only the
// position reading logic runs (no homing operations are executed).
// ============================================================================
IF currentState <> E_MachineState.HOMING THEN
    fbHome(
        Enable := FALSE,
        HomeAxis := 0,
        HomeSingle := FALSE,
        HomeAll := FALSE,
        Velocity := 10.0,
        Acceleration := 5.0,
        Deceleration := 5.0,
        Jerk := 10.0,
        PositionTolerance := 0.5
    );
    // Update isHomed based on actual axis positions (always, not just in HOMING mode)
    isHomed := fbHome.AllAxesHomed;
END_IF

// Copy internal state to output so other code/HMI can see it
currentStateOut := currentState;

// Copy motion sequence visualization outputs
motionStepOut := motionSequence.CurrentStep;
motionSubStep := motionSequence.CurrentSubStep;
activeAxesMask := motionSequence.ActiveAxesMask;
stepDescription := motionSequence.StepDescription;

// ============================================================================
// CENTRALIZED ERROR HANDLER
// Collects errors from all subsystems: power, gearing, and motion steps
// ============================================================================

errorHandler(
    // Reset errors when entering RESETTING state or when reset button pressed in FAULTED
    Reset := (currentState = E_MachineState.RESETTING),

    // Motion step errors (from FB_MotionSequence)
    LoadBlockError := motionSequence.LoadBlockError,
    LoadBlockErrorAxisID := motionSequence.LoadBlockErrorAxisID,
    CuttingError := motionSequence.CuttingError,           // G-Code cutting (replaces Studs/Windows/Electrical)
    CuttingErrorAxisID := motionSequence.CuttingErrorAxisID,
    UnloadError := motionSequence.UnloadError,
    UnloadErrorAxisID := motionSequence.UnloadErrorAxisID,

//     // Power errors (from FB_AxisPower via inputs)
//     PowerError := powerError,
//     PowerErrorAxisID := powerErrorAxisID,

    // Context for error reporting
    CurrentStep := motionSequence.CurrentStep,
    CurrentSubStep := motionSequence.CurrentSubStep
);

// Copy error handler outputs
hasError := errorHandler.HasError;
errorInfo := errorHandler.ErrorInfo;
errorCount := errorHandler.ErrorCount;
