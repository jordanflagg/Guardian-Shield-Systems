{attribute 'no_assign'}
FUNCTION_BLOCK FB_MotionSequence
(*
==============================================================================
  FB_MotionSequence
  Motion sequence controller — G-Code driven cutting

  Orchestrates the production cycle:
    WAIT_FOR_LOAD → SHIFT_TO_PROCESSING → CUTTING → SHIFT_TO_UNLOAD

  Position setpoints and motion parameters come from GVL_Machine.
  Cutting path comes from GVL_GCode (written by Python pipeline).

  Error handling: each step FB calls FC_ReportFault directly when an axis
  error occurs. FB_FaultMonitor (in PLC_PRG) catches these and trips HasFault,
  which the state machine sees within one scan. No error propagation wiring
  needed in this FB.
==============================================================================
*)

VAR_INPUT
    Enable : BOOL;                      // TRUE when machine in RUNNING state
    ResetSequence : BOOL;               // TRUE only when the supervisor wants a hard reset
    BlockLoaded : BOOL;                 // TRUE when the next foam block is physically in place
    FilesReady : BOOL;                  // TRUE when the Python pipeline staged G-code for the block
END_VAR

VAR_OUTPUT
    SequenceComplete : BOOL;            // Full cycle completed
    CurrentStep : E_MotionStep;         // Current motion phase

    // HMI visualization outputs
    CurrentSubStep : UINT;              // Current substep within motion step
    ActiveAxesMask : LWORD;             // Bitmask of currently moving axes
    StepDescription : STRING[80];       // Human-readable description
END_VAR

VAR
    motionStep  : E_MotionStep := E_MotionStep.WAIT_FOR_LOAD;

    stepLoadBlock : FB_MotionStep_LoadBlock;
    stepCutting   : FB_MotionStep_Cutting;
    stepUnload    : FB_MotionStep_Unload;
END_VAR

///////////////////////////////////////////////////////////////////////////////

// ============================================================================
// HANDLE RESET / DISABLE
// ============================================================================

IF ResetSequence THEN
    stepLoadBlock(Enable := FALSE);
    stepCutting(Enable := FALSE);
    stepUnload(Enable := FALSE);
    motionStep := E_MotionStep.WAIT_FOR_LOAD;
    SequenceComplete := FALSE;
    CurrentStep := motionStep;
    CurrentSubStep := 0;
    ActiveAxesMask := 0;
    StepDescription := 'Waiting for block to load';
    RETURN;
END_IF

IF NOT Enable THEN
    CurrentStep := motionStep;
    RETURN;
END_IF

// ============================================================================
// MOTION SEQUENCE STATE MACHINE
// ============================================================================

CASE motionStep OF

    // ────────────────────────────────────────────────────────────────────────
    E_MotionStep.WAIT_FOR_LOAD:
        SequenceComplete := FALSE;
        CurrentSubStep := 0;
        ActiveAxesMask := 0;

        stepLoadBlock(Enable := FALSE);
        stepCutting(Enable := FALSE);
        stepUnload(Enable := FALSE);

        // Both conditions must be true before advancing:
        //   blockLoaded    — operator/sensor confirms foam block is on the table
        //   bFilesReady    — Python pipeline has uploaded G-code for THIS block
        IF BlockLoaded AND FilesReady THEN
            GVL_GCode.sStatus := '';
            motionStep := E_MotionStep.SHIFT_TO_PROCESSING_AREA;
        ELSIF BlockLoaded AND NOT FilesReady THEN
            StepDescription := 'Waiting for G-code upload';
            GVL_GCode.sStatus := 'Block loaded - upload G-code for next block via pipeline';
        ELSE
            StepDescription := 'Waiting for block to load';
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    E_MotionStep.SHIFT_TO_PROCESSING_AREA:
        stepLoadBlock(
            Enable       := TRUE,
            Positions    := GVL_Machine.Positions,
            MotionParams := GVL_Machine.MotionParams
        );

        CurrentSubStep  := stepLoadBlock.CurrentSubStep;
        ActiveAxesMask  := stepLoadBlock.ActiveAxesMask;
        StepDescription := stepLoadBlock.StepDescription;

        IF stepLoadBlock.Done THEN
            stepLoadBlock(Enable := FALSE);
            motionStep := E_MotionStep.CUTTING;
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    E_MotionStep.CUTTING:
        stepCutting(Enable := TRUE);

        CurrentSubStep  := stepCutting.CurrentSubStep;
        ActiveAxesMask  := stepCutting.ActiveAxesMask;
        StepDescription := stepCutting.StepDescription;

        IF stepCutting.Done THEN
            stepCutting(Enable := FALSE);
            motionStep := E_MotionStep.SHIFT_TO_UNLOAD_AREA;
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    E_MotionStep.SHIFT_TO_UNLOAD_AREA:
        stepUnload(
            Enable       := TRUE,
            Positions    := GVL_Machine.Positions,
            MotionParams := GVL_Machine.MotionParams
        );

        CurrentSubStep  := stepUnload.CurrentSubStep;
        ActiveAxesMask  := stepUnload.ActiveAxesMask;
        StepDescription := stepUnload.StepDescription;

        IF stepUnload.Done THEN
            stepUnload(Enable := FALSE);
            SequenceComplete := TRUE;
            motionStep := E_MotionStep.WAIT_FOR_LOAD;
        END_IF

END_CASE

CurrentStep := motionStep;
