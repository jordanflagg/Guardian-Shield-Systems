{attribute 'no_assign'}
FUNCTION_BLOCK FB_MotionStep_Cutting
(*
==============================================================================
  FB_MotionStep_Cutting  v4.0 — PLC monitors, Python dispatches

  The PLC controls WHEN cutting runs (by setting bBusy when it reaches the
  cutting step).  Python (trigger_dispatch) owns HOW: it watches bBusy, fires
  the REST API cmd/file calls for each wire, monitors kinematic state, then
  sets bJobDone when all wires are back at STANDBY.

  PLC role (this FB):
    1. Wait for bFilesReady (Python API has staged G-code)
    2. Set bBusy = TRUE  →  Python sees this and dispatches wires
    3. Wait for bJobDone = TRUE  →  Python has confirmed all wires done
    4. Clear flags, signal Done to FB_MotionSequence

  No Data Layer write calls — dispatch is handled entirely by Python.
==============================================================================
*)

VAR_INPUT
    Enable : BOOL;
END_VAR

VAR_OUTPUT
    Done             : BOOL;
    Busy             : BOOL;
    Error            : BOOL;
    ErrorAxisID      : UINT;
    CurrentSubStep   : UINT;
    ActiveAxesMask   : LWORD;
    StepDescription  : STRING[80];
    CurrentMoveIndex : INT;
    TotalMoves       : INT;
END_VAR

VAR
    base    : FB_MotionStepBase;
    subStep : UINT;
END_VAR

///////////////////////////////////////////////////////////////////////////////

base(Enable := Enable);

IF base.ResetSubStep THEN
    subStep     := 0;
    Error       := FALSE;
    ErrorAxisID := 0;
END_IF

IF NOT base.ShouldRun THEN
    Done           := base.Done;
    Busy           := base.Busy;
    ActiveAxesMask := 0;
    CurrentSubStep := 0;
    RETURN;
END_IF

CurrentSubStep := subStep;

CASE subStep OF

    // ────────────────────────────────────────────────────────────────────────
    0: // Wait for Python API to finish uploading G-code
        StepDescription := 'Waiting for G-code files to be staged';
        ActiveAxesMask  := 0;
        Busy            := TRUE;

        IF GVL_GCode.bFilesReady THEN
            // Signal Python trigger service to dispatch wires
            GVL_GCode.bBusy          := TRUE;
            GVL_GCode.nPipelineStage := 5;
            GVL_GCode.sStatus        := 'Cutting in progress';
            StepDescription          := 'Cutting in progress';
            ActiveAxesMask           := 16#0000_0000_0FFF_C000;
            subStep                  := 10;
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    10: // Wait for Python to confirm all wires complete (bJobDone)
        IF GVL_GCode.bJobDone THEN
            subStep := 20;
        ELSIF GVL_GCode.bError THEN
            // Python reported an error — log but do not fault machine
            GVL_GCode.sStatus := 'Wire error reported — check Python logs';
            subStep := 20;  // Still advance so machine can be reset manually
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    20: // Complete
        StepDescription          := 'Cutting complete';
        ActiveAxesMask           := 0;
        GVL_GCode.bBusy          := FALSE;
        GVL_GCode.bFilesReady    := FALSE;
        GVL_GCode.nPipelineStage := 6;
        GVL_GCode.sStatus        := 'Cutting complete — ready for next block';
        Done := TRUE;
        Busy := FALSE;

END_CASE
