{attribute 'no_assign'}
FUNCTION_BLOCK FB_MotionStep_Cutting
(*
==============================================================================
  FB_MotionStep_Cutting  v8.0 — Always-Enabled Kinematics

  Wire1-4 and Wire5_6 kinematic groups are always enabled by FB_KinematicEnable
  (enabled once on DrivesReady rising edge, never disabled between groups).
  No KinReady handshake flags — trigger_dispatch.py dispatches immediately.

  Home move is the last line of each G-code cutting script — no separate
  home dispatch needed after each group completes.

  This FB's only job:
    1. Wait for bFilesReady (Python API has staged G-code)
    2. Set bBusy = TRUE  →  trigger_dispatch.py sees this and dispatches
    3. Wait for bJobDone = TRUE (Python confirms all groups complete)
       OR bError = TRUE (Python reports a script fault)
    4. Clear flags, signal Done to parent

  FB_KinematicEnable is called from PLC_PRG (not here) so its 24/7 opstate
  monitor runs regardless of machine state.

  Errors are reported directly to GVL_Faults via FC_ReportFault.
==============================================================================
*)

VAR_INPUT
    Enable : BOOL;
END_VAR

VAR_OUTPUT
    Done             : BOOL;
    Busy             : BOOL;
    CurrentSubStep   : UINT;
    ActiveAxesMask   : LWORD;
    StepDescription  : STRING[80];
END_VAR

VAR
    subStep     : UINT;
    stepStarted : BOOL;
    localError  : BOOL;     // Halts state machine; cleared on next Enable := FALSE
END_VAR

///////////////////////////////////////////////////////////////////////////////

IF NOT Enable THEN
    Done := FALSE; Busy := FALSE;
    ActiveAxesMask := 0; CurrentSubStep := 0;
    subStep := 0; stepStarted := FALSE; localError := FALSE;
    RETURN;
END_IF

IF NOT stepStarted THEN
    stepStarted := TRUE;
    Busy := TRUE;
    subStep := 0;
    localError := FALSE;
END_IF

CurrentSubStep := subStep;

IF localError THEN
    Busy := FALSE;
    ActiveAxesMask := 0;
    RETURN;
END_IF

CASE subStep OF

    // ────────────────────────────────────────────────────────────────────────
    0: // Wait for Python API to finish uploading G-code
        StepDescription := 'Waiting for G-code files to be staged';
        ActiveAxesMask  := 0;
        Busy            := TRUE;

        IF GVL_GCode.bFilesReady THEN
            GVL_GCode.bBusy          := TRUE;
            GVL_GCode.nPipelineStage := 5;
            GVL_GCode.sStatus        := 'Cutting in progress';
            StepDescription          := 'Cutting in progress';
            // Mask covers all axes that may move across any group:
            //   Bits 13-28 = axes 14-29 (studs)
            //   Bits 29-32 = axes 30-33 (Wire 5)
            //   Bits 34-35 = axes 35-36 (Wire 6 trough cutters)
            ActiveAxesMask           := 16#0000_000D_FFFF_E000;
            subStep                  := 10;
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    10: // Wait for FB_KinematicEnable (in PLC_PRG) to sequence all groups
        // FB_KinematicEnable watches bGroupX_KinReady / bGroupX_Done with Python.
        // Monitor its Done flag via GVL — it signals completion when all groups
        // have cut, homed, and been disabled.
        //
        // The Done signal is read from GVL_GCode.bJobDone (set by Python after
        // the last group) — FB_KinematicEnable.Done is wired through GVL_HMI
        // by PLC_PRG for HMI visibility, but the real completion signal here is
        // bJobDone (Python confirmation all wires returned home).
        IF GVL_GCode.bJobDone THEN
            subStep := 20;
        ELSIF GVL_GCode.bError THEN
            GVL_GCode.sStatus := 'Wire cutting error — check Python logs';
            GVL_GCode.bBusy   := FALSE;
            // bFilesReady intentionally preserved — STEP file and cuts unchanged.
            FC_ReportFault(
                source        := CONCAT('FB_MotionStep_Cutting/', GVL_GCode.sErrorWire),
                axisID        := 0,
                errorCode     := DWORD#0,
                description   := GVL_GCode.sErrorMessage,
                motionStep    := E_MotionStep.CUTTING,
                motionSubStep := subStep
            );
            localError := TRUE;
            Busy       := FALSE;
        END_IF

    // ────────────────────────────────────────────────────────────────────────
    20: // Complete
        StepDescription          := 'Cutting complete';
        ActiveAxesMask           := 0;
        GVL_GCode.bBusy          := FALSE;
        GVL_GCode.bFilesReady    := FALSE;
        GVL_GCode.nPipelineStage := 6;
        GVL_GCode.sStatus        := 'Cutting complete — ready for next block';
        Done := TRUE;
        Busy := FALSE;

END_CASE
