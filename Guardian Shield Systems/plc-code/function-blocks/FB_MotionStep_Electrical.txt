FUNCTION_BLOCK FB_MotionStep_Electrical
// Handles ELECTRICAL motion sequence
// Controls electrical trough cutting axes (35, 36 independent, 30 for positioning)
// 6 substeps total

VAR_INPUT
    Enable : BOOL;                      // TRUE to run this step
    Positions : ST_PositionSetpoints;   // Position setpoints from recipe
    MotionParams : ST_MotionParameters; // Motion parameters from recipe
END_VAR

VAR_OUTPUT
    Done : BOOL;                        // Step completed
    Busy : BOOL;                        // Step in progress
    Error : BOOL;                       // Step failed
    ErrorAxisID : UINT;                 // Which axis caused error
    CurrentSubStep : INT;               // Current substep (0-5)
    ActiveAxesMask : LWORD;             // Bitmask of moving axes (64-bit for 36 axes)
    StepDescription : STRING[80];       // Human-readable description
END_VAR

VAR
    subStep : INT := 0;

    // Electrical trough axes
    moverAxis30 : FB_AxisMover;         // Horizontal positioning
    moverAxis35 : FB_AxisMover;         // Electrical cutting (independent)
    moverAxis36 : FB_AxisMover;         // Electrical cutting (independent)

    // State tracking
    stepStarted : BOOL := FALSE;
END_VAR

///////////////////////////////////////////////////////////////////////////////

// Not enabled - reset state for next cycle
IF NOT Enable THEN
    Busy := FALSE;
    Done := FALSE;
    Error := FALSE;
    ErrorAxisID := 0;
    ActiveAxesMask := 0;
    stepStarted := FALSE;
    subStep := 0;
    RETURN;
END_IF

// Start on enable
IF Enable AND NOT stepStarted THEN
    stepStarted := TRUE;
    Busy := TRUE;
    subStep := 0;
END_IF

// Update output
CurrentSubStep := subStep;

// Substep state machine
CASE subStep OF

    // ----------------------------------------------------------------
    0: // Axes 30,32 move to starting position for trough
        StepDescription := 'Electrical - positioning for trough';

        moverAxis30(
            Execute := TRUE,
            AxisRef := Axis_30,
            Position := Positions.axis30_TargetPosition_3,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        ActiveAxesMask := SHL(LWORD#1, 29);  // Axis 30

        IF moverAxis30.Done THEN
            moverAxis30(Execute := FALSE);
            subStep := 1;
        ELSIF moverAxis30.Error THEN
            Error := TRUE;
            ErrorAxisID := 30;
        END_IF

    // ----------------------------------------------------------------
    1: // Axes 35,36 extend into foam to depth
        StepDescription := 'Electrical - extending into foam';

        moverAxis35(
            Execute := TRUE,
            AxisRef := Axis_35,
            Position := Positions.axis35_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        moverAxis36(
            Execute := TRUE,
            AxisRef := Axis_36,
            Position := Positions.axis36_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        ActiveAxesMask := SHL(LWORD#1, 34) OR SHL(LWORD#1, 35);  // Axes 35, 36

        IF moverAxis35.Done AND moverAxis36.Done THEN
            moverAxis35(Execute := FALSE);
            moverAxis36(Execute := FALSE);
            subStep := 2;
        ELSIF moverAxis35.Error OR moverAxis36.Error THEN
            Error := TRUE;
            IF moverAxis35.Error THEN ErrorAxisID := 35;
            ELSE ErrorAxisID := 36;
            END_IF
        END_IF

    // ----------------------------------------------------------------
    2: // Axes 30,32 move to cut trough
        StepDescription := 'Electrical - cutting trough';

        moverAxis30(
            Execute := TRUE,
            AxisRef := Axis_30,
            Position := Positions.axis30_TargetPosition_4,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        ActiveAxesMask := SHL(LWORD#1, 29);  // Axis 30

        IF moverAxis30.Done THEN
            moverAxis30(Execute := FALSE);
            subStep := 3;
        ELSIF moverAxis30.Error THEN
            Error := TRUE;
            ErrorAxisID := 30;
        END_IF

    // ----------------------------------------------------------------
    3: // Axes 35,36 retract from foam
        StepDescription := 'Electrical - retracting';

        moverAxis35(
            Execute := TRUE,
            AxisRef := Axis_35,
            Position := Positions.axis35_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        moverAxis36(
            Execute := TRUE,
            AxisRef := Axis_36,
            Position := Positions.axis36_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        ActiveAxesMask := SHL(LWORD#1, 34) OR SHL(LWORD#1, 35);  // Axes 35, 36

        IF moverAxis35.Done AND moverAxis36.Done THEN
            moverAxis35(Execute := FALSE);
            moverAxis36(Execute := FALSE);
            subStep := 4;
        ELSIF moverAxis35.Error OR moverAxis36.Error THEN
            Error := TRUE;
            IF moverAxis35.Error THEN ErrorAxisID := 35;
            ELSE ErrorAxisID := 36;
            END_IF
        END_IF

    // ----------------------------------------------------------------
    4: // All axes 30,35,36 return home
        StepDescription := 'Electrical - returning home';

        moverAxis30(
            Execute := TRUE,
            AxisRef := Axis_30,
            Position := Positions.axis30_HomePosition,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        moverAxis35(
            Execute := TRUE,
            AxisRef := Axis_35,
            Position := Positions.axis35_HomePosition,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        moverAxis36(
            Execute := TRUE,
            AxisRef := Axis_36,
            Position := Positions.axis36_HomePosition,
            Velocity := MotionParams.defaultVelocity,
            Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration,
            Jerk := MotionParams.defaultJerk
        );

        ActiveAxesMask := SHL(LWORD#1, 29) OR SHL(LWORD#1, 34) OR SHL(LWORD#1, 35);  // Axes 30, 35, 36

        IF moverAxis30.Done AND moverAxis35.Done AND moverAxis36.Done THEN
            moverAxis30(Execute := FALSE);
            moverAxis35(Execute := FALSE);
            moverAxis36(Execute := FALSE);
            subStep := 5;
        ELSIF moverAxis30.Error OR moverAxis35.Error OR moverAxis36.Error THEN
            Error := TRUE;
            IF moverAxis30.Error THEN ErrorAxisID := 30;
            ELSIF moverAxis35.Error THEN ErrorAxisID := 35;
            ELSE ErrorAxisID := 36;
            END_IF
        END_IF

    // ----------------------------------------------------------------
    5: // Complete - electrical trough done
        StepDescription := 'Electrical trough complete';
        ActiveAxesMask := 0;
        Done := TRUE;
        Busy := FALSE;

END_CASE

// Handle errors
IF Error THEN
    Busy := FALSE;
    ActiveAxesMask := 0;
END_IF
