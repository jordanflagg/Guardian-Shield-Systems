{attribute 'no_assign'}
FUNCTION_BLOCK FB_MotionStep_LoadBlock
// ============================================================================
// Handles SHIFT_TO_PROCESSING_AREA motion sequence
// Moves block from load table to processing center
//
// Axes: 1 (shift), 2 (load lift), 6 (center), 8 (pusher), 10 (unload lift)
// 12 substeps: sequential single-axis moves with geared slaves
//
// Errors are reported directly to GVL_Faults via FC_ReportFault — no error
// outputs to wire through parent FBs. localError halts the state machine
// in-place; FB_FaultMonitor trips the machine to FAULTED within one scan,
// then Enable goes FALSE and the FB resets cleanly.
// ============================================================================

VAR_INPUT
    Enable : BOOL;                      // TRUE to run this step
    Positions : ST_PositionSetpoints;   // Position setpoints from recipe
    MotionParams : ST_MotionParameters; // Motion parameters from recipe
END_VAR

VAR_OUTPUT
    Done : BOOL;                        // Step completed
    Busy : BOOL;                        // Step in progress
    CurrentSubStep : UINT;              // Current substep (0-11)
    ActiveAxesMask : LWORD;             // Bitmask of moving axes
    StepDescription : STRING[80];       // Human-readable description
END_VAR

VAR
    moverAxis1  : FB_AxisMover;
    moverAxis2  : FB_AxisMover;
    moverAxis6  : FB_AxisMover;
    moverAxis8  : FB_AxisMover;
    moverAxis10 : FB_AxisMover;
    subStep     : UINT;
    stepStarted : BOOL;
    localError  : BOOL;                 // Halts state machine; cleared on next Enable := FALSE
END_VAR

///////////////////////////////////////////////////////////////////////////////

IF NOT Enable THEN
    Done := FALSE; Busy := FALSE;
    ActiveAxesMask := 0; CurrentSubStep := 0; StepDescription := '';
    subStep := 0; stepStarted := FALSE; localError := FALSE;
    RETURN;
END_IF

IF NOT stepStarted THEN
    stepStarted := TRUE;
    Busy := TRUE;
    subStep := 0;
END_IF

CurrentSubStep := subStep;

// Hold state machine when a fault has been reported — wait for state machine
// to disable us via Enable := FALSE (happens within ~2 PLC scans via HasFault)
IF localError THEN
    Busy := FALSE;
    ActiveAxesMask := 0;
    RETURN;
END_IF

// Substep state machine
CASE subStep OF

    // =========================================================================
    0: // Axis 1 shifts block to left wall
    // =========================================================================
        StepDescription := 'Shifting block to left wall';
        ActiveAxesMask := FC_BuildAxisMask(1, 0, 0, 0, 0, 0, 0, 0);

        moverAxis1(Execute := TRUE, AxisRef := Axis_1, Position := Positions.axis1_ShiftPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis1.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 1, moverAxis1.ErrorID, 'Axis 1 shift failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis1.Done THEN
            moverAxis1(Execute := FALSE);
            subStep := 1;
        END_IF

    // =========================================================================
    1: // Axis 1 returns to home position
    // =========================================================================
        StepDescription := 'Returning left wall';
        ActiveAxesMask := FC_BuildAxisMask(1, 0, 0, 0, 0, 0, 0, 0);

        moverAxis1(Execute := TRUE, AxisRef := Axis_1, Position := Positions.axis1_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis1.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 1, moverAxis1.ErrorID, 'Axis 1 return failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis1.Done THEN
            moverAxis1(Execute := FALSE);
            subStep := 2;
        END_IF

    // =========================================================================
    2: // Axis 2 raises (lifts block off load table) - slaves 3,4,5 follow
    // =========================================================================
        StepDescription := 'Raising load table';
        ActiveAxesMask := FC_BuildAxisMask(2, 0, 0, 0, 0, 0, 0, 0);

        moverAxis2(Execute := TRUE, AxisRef := Axis_2, Position := Positions.axis2_RaisePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis2.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 2, moverAxis2.ErrorID, 'Axis 2 raise failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis2.Done THEN
            moverAxis2(Execute := FALSE);
            subStep := 3;
        END_IF

    // =========================================================================
    3: // Axis 6 shifts block toward center - slave 7 follows
    // =========================================================================
        StepDescription := 'Shifting block to center';
        ActiveAxesMask := FC_BuildAxisMask(6, 0, 0, 0, 0, 0, 0, 0);

        moverAxis6(Execute := TRUE, AxisRef := Axis_6, Position := Positions.axis6_ShiftPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis6.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 6, moverAxis6.ErrorID, 'Axis 6 shift failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis6.Done THEN
            moverAxis6(Execute := FALSE);
            subStep := 4;
        END_IF

    // =========================================================================
    4: // Axis 2 lowers (back down below table)
    // =========================================================================
        StepDescription := 'Lowering load table';
        ActiveAxesMask := FC_BuildAxisMask(2, 0, 0, 0, 0, 0, 0, 0);

        moverAxis2(Execute := TRUE, AxisRef := Axis_2, Position := Positions.axis2_LowerPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis2.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 2, moverAxis2.ErrorID, 'Axis 2 lower failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis2.Done THEN
            moverAxis2(Execute := FALSE);
            subStep := 5;
        END_IF

    // =========================================================================
    5: // Axis 6 returns to home position
    // =========================================================================
        StepDescription := 'Returning shift mechanism';
        ActiveAxesMask := FC_BuildAxisMask(6, 0, 0, 0, 0, 0, 0, 0);

        moverAxis6(Execute := TRUE, AxisRef := Axis_6, Position := Positions.axis6_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis6.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 6, moverAxis6.ErrorID, 'Axis 6 return failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis6.Done THEN
            moverAxis6(Execute := FALSE);
            subStep := 6;
        END_IF

    // =========================================================================
    6: // Axis 10 raises (prepare to receive block) - slaves 11,12,13 follow
    // =========================================================================
        StepDescription := 'Raising unload table';
        ActiveAxesMask := FC_BuildAxisMask(10, 0, 0, 0, 0, 0, 0, 0);

        moverAxis10(Execute := TRUE, AxisRef := Axis_10, Position := Positions.axis10_RaisePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis10.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 10, moverAxis10.ErrorID, 'Axis 10 raise failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis10.Done THEN
            moverAxis10(Execute := FALSE);
            subStep := 7;
        END_IF

    // =========================================================================
    7: // Axis 8 shifts block to center - slave 9 follows
    // =========================================================================
        StepDescription := 'Centering block';
        ActiveAxesMask := FC_BuildAxisMask(8, 0, 0, 0, 0, 0, 0, 0);

        moverAxis8(Execute := TRUE, AxisRef := Axis_8, Position := Positions.axis8_ShiftPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis8.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 8, moverAxis8.ErrorID, 'Axis 8 shift failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis8.Done THEN
            moverAxis8(Execute := FALSE);
            subStep := 8;
        END_IF

    // =========================================================================
    8: // Wait for suction cups to enable
    // =========================================================================
        StepDescription := 'Engaging suction cups';
        ActiveAxesMask := 0;

        // TODO: Add suction cup I/O when available
        // For now, proceed immediately
        subStep := 9;

    // =========================================================================
    9: // Axis 8 returns to home position
    // =========================================================================
        StepDescription := 'Returning pusher';
        ActiveAxesMask := FC_BuildAxisMask(8, 0, 0, 0, 0, 0, 0, 0);

        moverAxis8(Execute := TRUE, AxisRef := Axis_8, Position := Positions.axis8_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis8.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 8, moverAxis8.ErrorID, 'Axis 8 return failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis8.Done THEN
            moverAxis8(Execute := FALSE);
            subStep := 10;
        END_IF

    // =========================================================================
    10: // Axis 10 lowers (get out of way)
    // =========================================================================
        StepDescription := 'Lowering unload table';
        ActiveAxesMask := FC_BuildAxisMask(10, 0, 0, 0, 0, 0, 0, 0);

        moverAxis10(Execute := TRUE, AxisRef := Axis_10, Position := Positions.axis10_LowerPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis10.Error THEN
            FC_ReportFault('FB_MotionStep_LoadBlock', 10, moverAxis10.ErrorID, 'Axis 10 lower failed', E_MotionStep.SHIFT_TO_PROCESSING_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis10.Done THEN
            moverAxis10(Execute := FALSE);
            subStep := 11;
        END_IF

    // =========================================================================
    11: // Complete - block is now in center, ready for carving
    // =========================================================================
        StepDescription := 'Block positioned - ready for carving';
        ActiveAxesMask := 0;
        Done := TRUE;
        Busy := FALSE;

END_CASE
