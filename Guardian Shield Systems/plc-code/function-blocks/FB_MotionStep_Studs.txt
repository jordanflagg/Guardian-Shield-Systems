{attribute 'no_assign'}
FUNCTION_BLOCK FB_MotionStep_Studs
// ============================================================================
// Handles STUDS motion sequence - Carves studs using 8 pairs (16 axes total)
// Uses FB_MotionStepBase for enable/disable handling
// Uses utility functions for cleaner status aggregation
//
// Axes: 14,15,18,19,22,23,26,27 (masters) with 16,17,20,21,24,25,28,29 (slaves)
// 6 substeps: position -> extend -> cut -> retract -> home -> complete
// ============================================================================

VAR_INPUT
    Enable : BOOL;                      // TRUE to run this step
    Positions : ST_PositionSetpoints;   // Position setpoints from recipe
    MotionParams : ST_MotionParameters; // Motion parameters from recipe
END_VAR

VAR_OUTPUT
    Done : BOOL;                        // Step completed
    Busy : BOOL;                        // Step in progress
    Error : BOOL;                       // Step failed
    ErrorAxisID : UNIT;                 // Which axis caused error
    CurrentSubStep : UNIT;               // Current substep (0-5)
    ActiveAxesMask : LWORD;             // Bitmask of moving axes
    StepDescription : STRING[80];       // Human-readable description
END_VAR

VAR
    // Base handler for enable/disable logic
    base : FB_MotionStepBase;

    // Vertical stud axes (masters: 14, 18, 22, 26)
    moverAxis14 : FB_AxisMover;
    moverAxis18 : FB_AxisMover;
    moverAxis22 : FB_AxisMover;
    moverAxis26 : FB_AxisMover;

    // Horizontal stud axes (masters: 15, 19, 23, 27)
    moverAxis15 : FB_AxisMover;
    moverAxis19 : FB_AxisMover;
    moverAxis23 : FB_AxisMover;
    moverAxis27 : FB_AxisMover;

    // Status structures for utility functions
    verticalStatus : ARRAY[1..4] OF ST_MoverStatus;
    horizontalStatus : ARRAY[1..4] OF ST_MoverStatus;
    allStatus : ARRAY[1..8] OF ST_MoverStatus;

    // Completion tracking
    allVerticalDone : BOOL;
    allHorizontalDone : BOOL;
    tempErrorAxis : UINT;

    // Local substep tracking
    subStep : UNIT;

    // Empty placeholder for unused mover slots
    emptyStatus : ST_MoverStatus := (Done := FALSE, Error := FALSE, AxisID := 0);
END_VAR

///////////////////////////////////////////////////////////////////////////////

// Run base enable/disable handler
base(Enable := Enable);

// Reset substep on new enable
IF base.ResetSubStep THEN
    subStep := 0;
END_IF

// Handle disabled state
IF NOT base.ShouldRun THEN
    Done := base.Done;
    Busy := base.Busy;
    Error := FALSE;
    ErrorAxisID := 0;
    ActiveAxesMask := 0;
    CurrentSubStep := 0;
    StepDescription := '';
    RETURN;
END_IF

// Update outputs
CurrentSubStep := subStep;

// Build status arrays for utility functions
verticalStatus[1].Done := moverAxis14.Done;
verticalStatus[1].Error := moverAxis14.Error;
verticalStatus[1].AxisID := 14;

verticalStatus[2].Done := moverAxis18.Done;
verticalStatus[2].Error := moverAxis18.Error;
verticalStatus[2].AxisID := 18;

verticalStatus[3].Done := moverAxis22.Done;
verticalStatus[3].Error := moverAxis22.Error;
verticalStatus[3].AxisID := 22;

verticalStatus[4].Done := moverAxis26.Done;
verticalStatus[4].Error := moverAxis26.Error;
verticalStatus[4].AxisID := 26;

horizontalStatus[1].Done := moverAxis15.Done;
horizontalStatus[1].Error := moverAxis15.Error;
horizontalStatus[1].AxisID := 15;

horizontalStatus[2].Done := moverAxis19.Done;
horizontalStatus[2].Error := moverAxis19.Error;
horizontalStatus[2].AxisID := 19;

horizontalStatus[3].Done := moverAxis23.Done;
horizontalStatus[3].Error := moverAxis23.Error;
horizontalStatus[3].AxisID := 23;

horizontalStatus[4].Done := moverAxis27.Done;
horizontalStatus[4].Error := moverAxis27.Error;
horizontalStatus[4].AxisID := 27;

// Substep state machine
CASE subStep OF

    // =========================================================================
    0: // Vertical axes move to position 1 (14,18,22,26 with slaves 16,20,24,28)
    // =========================================================================
        StepDescription := 'Stud axes - vertical to position 1';
        ActiveAxesMask := FC_BuildAxisMask(14, 18, 22, 26, 0, 0, 0, 0);

        moverAxis14(Execute := TRUE, AxisRef := Axis_14, Position := Positions.axis14_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis18(Execute := TRUE, AxisRef := Axis_18, Position := Positions.axis18_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis22(Execute := TRUE, AxisRef := Axis_22, Position := Positions.axis22_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis26(Execute := TRUE, AxisRef := Axis_26, Position := Positions.axis26_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        // Check for errors using utility function
        tempErrorAxis := FC_FindFirstErrorAxis(verticalStatus[1], verticalStatus[2], verticalStatus[3], verticalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus);

        IF tempErrorAxis > 0 THEN
            Error := TRUE;
            ErrorAxisID := tempErrorAxis;
        ELSIF FC_CheckMoversComplete(verticalStatus[1], verticalStatus[2], verticalStatus[3], verticalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus) THEN
            // Clear movers and advance
            moverAxis14(Execute := FALSE); moverAxis18(Execute := FALSE);
            moverAxis22(Execute := FALSE); moverAxis26(Execute := FALSE);
            subStep :=1;
        END_IF

    // =========================================================================
    1: // Horizontal axes extend to position 1 (15,19,23,27 with slaves 17,21,25,29)
    // =========================================================================
        StepDescription := 'Stud axes - horizontal extend';
        ActiveAxesMask := FC_BuildAxisMask(15, 19, 23, 27, 0, 0, 0, 0);

        moverAxis15(Execute := TRUE, AxisRef := Axis_15, Position := Positions.axis15_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis19(Execute := TRUE, AxisRef := Axis_19, Position := Positions.axis19_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis23(Execute := TRUE, AxisRef := Axis_23, Position := Positions.axis23_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis27(Execute := TRUE, AxisRef := Axis_27, Position := Positions.axis27_TargetPosition_1,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        tempErrorAxis := FC_FindFirstErrorAxis(horizontalStatus[1], horizontalStatus[2], horizontalStatus[3], horizontalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus);

        IF tempErrorAxis > 0 THEN
            Error := TRUE;
            ErrorAxisID := tempErrorAxis;
        ELSIF FC_CheckMoversComplete(horizontalStatus[1], horizontalStatus[2], horizontalStatus[3], horizontalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus) THEN
            moverAxis15(Execute := FALSE); moverAxis19(Execute := FALSE);
            moverAxis23(Execute := FALSE); moverAxis27(Execute := FALSE);
            subStep :=2;
        END_IF

    // =========================================================================
    2: // Vertical axes cut to position 2
    // =========================================================================
        StepDescription := 'Stud axes - cutting (vertical)';
        ActiveAxesMask := FC_BuildAxisMask(14, 18, 22, 26, 0, 0, 0, 0);

        moverAxis14(Execute := TRUE, AxisRef := Axis_14, Position := Positions.axis14_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis18(Execute := TRUE, AxisRef := Axis_18, Position := Positions.axis18_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis22(Execute := TRUE, AxisRef := Axis_22, Position := Positions.axis22_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis26(Execute := TRUE, AxisRef := Axis_26, Position := Positions.axis26_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        tempErrorAxis := FC_FindFirstErrorAxis(verticalStatus[1], verticalStatus[2], verticalStatus[3], verticalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus);

        IF tempErrorAxis > 0 THEN
            Error := TRUE;
            ErrorAxisID := tempErrorAxis;
        ELSIF FC_CheckMoversComplete(verticalStatus[1], verticalStatus[2], verticalStatus[3], verticalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus) THEN
            moverAxis14(Execute := FALSE); moverAxis18(Execute := FALSE);
            moverAxis22(Execute := FALSE); moverAxis26(Execute := FALSE);
            subStep :=3;
        END_IF

    // =========================================================================
    3: // Horizontal axes retract to position 2
    // =========================================================================
        StepDescription := 'Stud axes - retracting';
        ActiveAxesMask := FC_BuildAxisMask(15, 19, 23, 27, 0, 0, 0, 0);

        moverAxis15(Execute := TRUE, AxisRef := Axis_15, Position := Positions.axis15_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis19(Execute := TRUE, AxisRef := Axis_19, Position := Positions.axis19_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis23(Execute := TRUE, AxisRef := Axis_23, Position := Positions.axis23_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis27(Execute := TRUE, AxisRef := Axis_27, Position := Positions.axis27_TargetPosition_2,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        tempErrorAxis := FC_FindFirstErrorAxis(horizontalStatus[1], horizontalStatus[2], horizontalStatus[3], horizontalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus);

        IF tempErrorAxis > 0 THEN
            Error := TRUE;
            ErrorAxisID := tempErrorAxis;
        ELSIF FC_CheckMoversComplete(horizontalStatus[1], horizontalStatus[2], horizontalStatus[3], horizontalStatus[4],
            emptyStatus, emptyStatus, emptyStatus, emptyStatus) THEN
            moverAxis15(Execute := FALSE); moverAxis19(Execute := FALSE);
            moverAxis23(Execute := FALSE); moverAxis27(Execute := FALSE);
            subStep :=4;
        END_IF

    // =========================================================================
    4: // All stud axes return home (8 axes in parallel)
    // =========================================================================
        StepDescription := 'Stud axes - returning home';
        ActiveAxesMask := FC_BuildAxisMask(14, 15, 18, 19, 22, 23, 26, 27);

        // Vertical axes home
        moverAxis14(Execute := TRUE, AxisRef := Axis_14, Position := Positions.axis14_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis18(Execute := TRUE, AxisRef := Axis_18, Position := Positions.axis18_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis22(Execute := TRUE, AxisRef := Axis_22, Position := Positions.axis22_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis26(Execute := TRUE, AxisRef := Axis_26, Position := Positions.axis26_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        // Horizontal axes home
        moverAxis15(Execute := TRUE, AxisRef := Axis_15, Position := Positions.axis15_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis19(Execute := TRUE, AxisRef := Axis_19, Position := Positions.axis19_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis23(Execute := TRUE, AxisRef := Axis_23, Position := Positions.axis23_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);
        moverAxis27(Execute := TRUE, AxisRef := Axis_27, Position := Positions.axis27_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        // Build combined status for all 8 movers
        allStatus[1] := verticalStatus[1];   allStatus[2] := horizontalStatus[1];
        allStatus[3] := verticalStatus[2];   allStatus[4] := horizontalStatus[2];
        allStatus[5] := verticalStatus[3];   allStatus[6] := horizontalStatus[3];
        allStatus[7] := verticalStatus[4];   allStatus[8] := horizontalStatus[4];

        tempErrorAxis := FC_FindFirstErrorAxis(allStatus[1], allStatus[2], allStatus[3], allStatus[4],
            allStatus[5], allStatus[6], allStatus[7], allStatus[8]);

        IF tempErrorAxis > 0 THEN
            Error := TRUE;
            ErrorAxisID := tempErrorAxis;
        ELSIF FC_CheckMoversComplete(allStatus[1], allStatus[2], allStatus[3], allStatus[4],
            allStatus[5], allStatus[6], allStatus[7], allStatus[8]) THEN
            // Clear all movers
            moverAxis14(Execute := FALSE); moverAxis15(Execute := FALSE);
            moverAxis18(Execute := FALSE); moverAxis19(Execute := FALSE);
            moverAxis22(Execute := FALSE); moverAxis23(Execute := FALSE);
            moverAxis26(Execute := FALSE); moverAxis27(Execute := FALSE);
            subStep :=5;
        END_IF

    // =========================================================================
    5: // Complete - stud carving done
    // =========================================================================
        StepDescription := 'Stud carving complete';
        ActiveAxesMask := 0;
        Done := TRUE;
        Busy := FALSE;

END_CASE

// Handle errors - stop motion and clear busy
IF Error THEN
    Busy := FALSE;
    ActiveAxesMask := 0;
END_IF
