{attribute 'no_assign'}
FUNCTION_BLOCK FB_MotionStep_Unload
// ============================================================================
// Handles SHIFT_TO_UNLOAD_AREA motion sequence
// Moves completed block from processing center to unload table
//
// Axes: 8 (pusher), 10 (table lift) with slaves 9, 11, 12, 13
// 5 substeps: raise -> shift -> lower -> return -> complete
//
// Errors are reported directly to GVL_Faults via FC_ReportFault — no error
// outputs to wire through parent FBs.
// ============================================================================

VAR_INPUT
    Enable : BOOL;                      // TRUE to run this step
    Positions : ST_PositionSetpoints;   // Position setpoints from recipe
    MotionParams : ST_MotionParameters; // Motion parameters from recipe
END_VAR

VAR_OUTPUT
    Done : BOOL;                        // Step completed
    Busy : BOOL;                        // Step in progress
    CurrentSubStep : UINT;              // Current substep (0-4)
    ActiveAxesMask : LWORD;             // Bitmask of moving axes
    StepDescription : STRING[80];       // Human-readable description
END_VAR

VAR
    moverAxis8  : FB_AxisMover;
    moverAxis10 : FB_AxisMover;
    subStep     : UINT;
    stepStarted : BOOL;
    localError  : BOOL;     // Halts state machine; cleared on next Enable := FALSE
END_VAR

///////////////////////////////////////////////////////////////////////////////

IF NOT Enable THEN
    // Drive movers to Execute := FALSE so MC_MoveAbsolute gets a rising edge on next enable.
    // Without this, fbMove inside FB_AxisMover stays latched Execute=TRUE; the next call with
    // Execute=TRUE produces no edge and reports the stale ErrorID (0x7FFF) as a phantom fault.
    moverAxis8(Execute := FALSE, AxisRef := Axis_8);
    moverAxis10(Execute := FALSE, AxisRef := Axis_10);
    Done := FALSE; Busy := FALSE;
    ActiveAxesMask := 0; CurrentSubStep := 0; StepDescription := '';
    subStep := 0; stepStarted := FALSE; localError := FALSE;
    RETURN;
END_IF

IF NOT stepStarted THEN
    stepStarted := TRUE;
    Busy := TRUE;
    subStep := 0;
END_IF

CurrentSubStep := subStep;

// Hold state machine on fault — wait for Enable := FALSE from parent chain
IF localError THEN
    Busy := FALSE;
    ActiveAxesMask := 0;
    RETURN;
END_IF

// Substep state machine
CASE subStep OF

    // =========================================================================
    0: // Axis 10 raises to pick up block - slaves 11,12,13 follow
    // =========================================================================
        StepDescription := 'Unload - raising table';
        ActiveAxesMask := FC_BuildAxisMask(10, 0, 0, 0, 0, 0, 0, 0);

        // TODO: Disable suction cups first when I/O is available

        moverAxis10(Execute := TRUE, AxisRef := Axis_10, Position := Positions.axis10_RaisePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis10.Error THEN
            FC_ReportFault('FB_MotionStep_Unload', 10, moverAxis10.ErrorID, 'Axis 10 raise failed', E_MotionStep.SHIFT_TO_UNLOAD_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis10.Done THEN
            moverAxis10(Execute := FALSE);
            subStep := 1;
        END_IF

    // =========================================================================
    1: // Axis 8 shifts block to unload table - slave 9 follows
    // =========================================================================
        StepDescription := 'Unload - shifting block';
        ActiveAxesMask := FC_BuildAxisMask(8, 0, 0, 0, 0, 0, 0, 0);

        moverAxis8(Execute := TRUE, AxisRef := Axis_8, Position := Positions.axis8_UnloadPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis8.Error THEN
            FC_ReportFault('FB_MotionStep_Unload', 8, moverAxis8.ErrorID, 'Axis 8 unload shift failed', E_MotionStep.SHIFT_TO_UNLOAD_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis8.Done THEN
            moverAxis8(Execute := FALSE);
            subStep := 2;
        END_IF

    // =========================================================================
    2: // Axis 10 lowers (place block on unload table)
    // =========================================================================
        StepDescription := 'Unload - lowering table';
        ActiveAxesMask := FC_BuildAxisMask(10, 0, 0, 0, 0, 0, 0, 0);

        moverAxis10(Execute := TRUE, AxisRef := Axis_10, Position := Positions.axis10_LowerPosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis10.Error THEN
            FC_ReportFault('FB_MotionStep_Unload', 10, moverAxis10.ErrorID, 'Axis 10 lower failed', E_MotionStep.SHIFT_TO_UNLOAD_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis10.Done THEN
            moverAxis10(Execute := FALSE);
            subStep := 3;
        END_IF

    // =========================================================================
    3: // Axis 8 returns home
    // =========================================================================
        StepDescription := 'Unload - returning pusher';
        ActiveAxesMask := FC_BuildAxisMask(8, 0, 0, 0, 0, 0, 0, 0);

        moverAxis8(Execute := TRUE, AxisRef := Axis_8, Position := Positions.axis8_HomePosition,
            Velocity := MotionParams.defaultVelocity, Acceleration := MotionParams.defaultAcceleration,
            Deceleration := MotionParams.defaultDeceleration, Jerk := MotionParams.defaultJerk);

        IF moverAxis8.Error THEN
            FC_ReportFault('FB_MotionStep_Unload', 8, moverAxis8.ErrorID, 'Axis 8 return failed', E_MotionStep.SHIFT_TO_UNLOAD_AREA, subStep);
            localError := TRUE;
        ELSIF moverAxis8.Done THEN
            moverAxis8(Execute := FALSE);
            subStep := 4;
        END_IF

    // =========================================================================
    4: // Complete - block on unload table, ready for fork truck
    // =========================================================================
        StepDescription := 'Cycle complete - ready for removal';
        ActiveAxesMask := 0;
        Done := TRUE;
        Busy := FALSE;

END_CASE
