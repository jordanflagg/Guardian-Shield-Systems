{attribute 'no_assign'}
FUNCTION_BLOCK FB_ScriptDispatch
(*
==============================================================================
  FB_ScriptDispatch
  Dispatches a G-code file to a ctrlX script instance via DL_WriteNode,
  OR resets a script instance (parameterless cmd/reset call).

  Dispatch:  calls script/instances/{WireName}/cmd/file  with plain STRING path
  Reset:     calls script/instances/{WireName}/cmd/reset with no data

  Edge-triggered: Execute rising edge fires the call once.
  Reset Execute to FALSE after Done/Error before re-using.
==============================================================================
*)

VAR_INPUT
    Execute    : BOOL;
    WireName   : STRING(20);        // e.g. 'Wire1'
    ScriptPath : STRING(255);       // Full .npg path — ignored when IsReset=TRUE
    IsReset    : BOOL;              // TRUE = cmd/reset, FALSE = cmd/file dispatch
END_VAR

VAR_OUTPUT
    Done  : BOOL;
    Busy  : BOOL;
    Error : BOOL;
END_VAR

VAR
    fbCall   : DL_WriteNode;
    nodePath : STRING(255);
    dummy    : BYTE := 0;       // Passed to DL_WriteNode for parameterless cmd/reset (ANY param)
END_VAR

///////////////////////////////////////////////////////////////////////////////

IF IsReset THEN
    // Script instance reset — no payload, pass dummy byte
    nodePath := CONCAT('script/instances/', CONCAT(WireName, '/cmd/reset'));

    fbCall(
        Execute  := Execute,
        NodeName := nodePath,
        Value    := dummy
    );
ELSE
    // File dispatch — ctrlX cmd/file accepts the script path as a plain STRING
    nodePath := CONCAT('script/instances/', CONCAT(WireName, '/cmd/file'));

    fbCall(
        Execute  := Execute,
        NodeName := nodePath,
        Value    := ScriptPath
    );
END_IF

Done  := fbCall.Done;
Busy  := fbCall.Active;
Error := fbCall.Error;
