FUNCTION_BLOCK FB_MotionSequence

VAR_INPUT
    enable : BOOL;
    recipeID : UINT := 1;           // Recipe number to load (1-10)
    loadRecipe : BOOL;              // Rising edge loads recipe
END_VAR

VAR_OUTPUT
    sequenceComplete : BOOL;
    currentStep : E_MotionStep;
    activeRecipeID : UINT;          // Currently loaded recipe ID
    recipeLoadComplete : BOOL;      // Recipe loaded successfully
    recipeLoadError : BOOL;         // Recipe load failed (invalid ID or inactive)

    // HMI visualization outputs
    currentSubStep : INT;           // Current substep within motion step
    activeAxesMask : DWORD;         // Bitmask of currently moving axes (bit 0 = axis 1, etc)
    stepDescription : STRING[80];   // Human-readable description of current action
END_VAR

VAR
    motionStep : E_MotionStep := E_MotionStep.WAIT_FOR_LOAD;
    subStep : INT := 0;

    positions : ST_PositionSetpoints;
    motionParams : ST_MotionParameters;

    // Recipe management
    currentRecipeID : UINT := 1;
    recipeLoaded : BOOL := FALSE;
    loadRecipePrev : BOOL := FALSE;     // For edge detection

    // Operation enable flags (from recipe)
    enableStudsOperation : BOOL := TRUE;
    enableWindowsDoorsOperation : BOOL := TRUE;
    enableElectricalOperation : BOOL := TRUE;

    blockLoaded : BOOL := TRUE;
    
    // Execute flags - ONLY for masters and independent axes
    executeAxis1Move : BOOL := FALSE;
    executeAxis2Move : BOOL := FALSE;   // Master for 3,4,5
    executeAxis6Move : BOOL := FALSE;   // Master for 7
    executeAxis8Move : BOOL := FALSE;   // Master for 9

    executeAxis14Move : BOOL := FALSE;  // Master for 16
    executeAxis15Move : BOOL := FALSE;  // Master for 17
    executeAxis18Move : BOOL := FALSE;  // Master for 20
    executeAxis19Move : BOOL := FALSE;  // Master for 21
    executeAxis22Move : BOOL := FALSE;  // Master for 24
    executeAxis23Move : BOOL := FALSE;  // Master for 25
    executeAxis26Move : BOOL := FALSE;  // Master for 28
    executeAxis27Move : BOOL := FALSE;  // Master for 29
    executeAxis30Move : BOOL := FALSE;  // Master for 32
    executeAxis31Move : BOOL := FALSE;  // Master for 33
    executeAxis34Move : BOOL := FALSE;  // Independent
    executeAxis35Move : BOOL := FALSE;  // Sometimes Master for 36
    executeAxis36Move : BOOL := FALSE;  // Sometimes Independent
    
    // Target positions - ONLY for masters and independent axes
    targetPosAxis1 : LREAL := 0.0;
    targetPosAxis2 : LREAL := 0.0;
    targetPosAxis6 : LREAL := 0.0;
    targetPosAxis8 : LREAL := 0.0;
    targetPosAxis10 : LREAL := 0.0;
    targetPosAxis14 : LREAL := 0.0;
    targetPosAxis15 : LREAL := 0.0;
    targetPosAxis18 : LREAL := 0.0;
    targetPosAxis19 : LREAL := 0.0;
    targetPosAxis22 : LREAL := 0.0;
    targetPosAxis23 : LREAL := 0.0;
    targetPosAxis26 : LREAL := 0.0;
    targetPosAxis27 : LREAL := 0.0;
    targetPosAxis30 : LREAL := 0.0;
    targetPosAxis31 : LREAL := 0.0;
    targetPosAxis34 : LREAL := 0.0;
    targetPosAxis35 : LREAL := 0.0;
    targetPosAxis36 : LREAL := 0.0;
    
    // Motion function blocks - ONLY for masters and independent axes
    fbMoveAxis1 : MC_MoveAbsolute;
    fbMoveAxis2 : MC_MoveAbsolute;
    fbMoveAxis6 : MC_MoveAbsolute;
    fbMoveAxis8 : MC_MoveAbsolute;
    fbMoveAxis10 : MC_MoveAbsolute;
    fbMoveAxis14 : MC_MoveAbsolute;
    fbMoveAxis15 : MC_MoveAbsolute;
    fbMoveAxis18 : MC_MoveAbsolute;
    fbMoveAxis19 : MC_MoveAbsolute;
    fbMoveAxis22 : MC_MoveAbsolute;
    fbMoveAxis23 : MC_MoveAbsolute;
    fbMoveAxis26 : MC_MoveAbsolute;
    fbMoveAxis27 : MC_MoveAbsolute;
    fbMoveAxis30 : MC_MoveAbsolute;
    fbMoveAxis31 : MC_MoveAbsolute;
    fbMoveAxis34 : MC_MoveAbsolute;
    fbMoveAxis35 : MC_MoveAbsolute;
    fbMoveAxis36 : MC_MoveAbsolute;
    
    // Gearing function blocks (permanent gearing)
    fbGearInAxis3 : MB_GearInPos;   // Axis 3 follows Axis 2
    fbGearInAxis4 : MB_GearInPos;   // Axis 4 follows Axis 2
    fbGearInAxis5 : MB_GearInPos;   // Axis 5 follows Axis 2
    fbGearInAxis7 : MB_GearInPos;   // Axis 7 follows Axis 6
    fbGearInAxis9 : MB_GearInPos;   // Axis 9 follows Axis 8
    fbGearInAxis11 : MB_GearInPos;  // Axis 11 follows Axis 10
    fbGearInAxis12 : MB_GearInPos;  // Axis 12 follows Axis 10
    fbGearInAxis13 : MB_GearInPos;  // Axis 13 follows Axis 10
    fbGearInAxis16 : MB_GearInPos;  // Axis 16 follows Axis 14
    fbGearInAxis17 : MB_GearInPos;  // Axis 17 follows Axis 15
    fbGearInAxis20 : MB_GearInPos;  // Axis 20 follows Axis 18
    fbGearInAxis21 : MB_GearInPos;  // Axis 21 follows Axis 19
    fbGearInAxis24 : MB_GearInPos;  // Axis 24 follows Axis 22
    fbGearInAxis25 : MB_GearInPos;  // Axis 25 follows Axis 23
    fbGearInAxis28 : MB_GearInPos;  // Axis 28 follows Axis 26
    fbGearInAxis29 : MB_GearInPos;  // Axis 29 follows Axis 27
    fbGearInAxis32 : MB_GearInPos;  // Axis 32 follows Axis 30
    fbGearInAxis33 : MB_GearInPos;  // Axis 33 follows Axis 31
    
END_VAR


///////////////////////////////////////////////////////////////////////////////

// ============================================================================
// RECIPE LOADING - Execute before motion sequence
// ============================================================================

// Load recipe on rising edge or if no recipe loaded yet
IF (loadRecipe AND NOT loadRecipePrev) OR NOT recipeLoaded THEN
    IF recipeID >= 1 AND recipeID <= 10 THEN
        IF GVL_Recipes.RecipeLibrary[recipeID].isActive THEN
            // Load recipe data
            positions := GVL_Recipes.RecipeLibrary[recipeID].positions;
            motionParams := GVL_Recipes.RecipeLibrary[recipeID].motionParams;
            enableStudsOperation := GVL_Recipes.RecipeLibrary[recipeID].enableStuds;
            enableWindowsDoorsOperation := GVL_Recipes.RecipeLibrary[recipeID].enableWindowsDoors;
            enableElectricalOperation := GVL_Recipes.RecipeLibrary[recipeID].enableElectrical;

            currentRecipeID := recipeID;
            recipeLoaded := TRUE;
            recipeLoadError := FALSE;
            recipeLoadComplete := TRUE;
            activeRecipeID := recipeID;
        ELSE
            // Recipe is inactive
            recipeLoadError := TRUE;
            recipeLoadComplete := FALSE;
        END_IF
    ELSE
        // Invalid recipe ID
        recipeLoadError := TRUE;
        recipeLoadComplete := FALSE;
    END_IF
END_IF

// Update edge detection
loadRecipePrev := loadRecipe;

// ============================================================================
// MOTION SEQUENCE STATE MACHINE
// ============================================================================

CASE motionStep OF
    
    E_MotionStep.WAIT_FOR_LOAD:
        sequenceComplete := FALSE;
        subStep := 0;
        
        IF blockLoaded THEN
            motionStep := E_MotionStep.SHIFT_TO_PROCESSING_AREA;
        END_IF;
    
    E_MotionStep.SHIFT_TO_PROCESSING_AREA:
    
    CASE subStep OF
        
        // ----------------------------------------------------------------
        0: // Step 2: Axis 1 shifts block to left wall
            targetPosAxis1 := positions.axis1_ShiftPosition;
            executeAxis1Move := TRUE;
            
            IF fbMoveAxis1.Done THEN
                executeAxis1Move := FALSE;
                subStep := 1;
            END_IF;
            
            IF fbMoveAxis1.Error THEN
                executeAxis1Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        1: // Step 3: Axes 2,3,4,5 raise (lift block off load table)
            targetPosAxis2 := positions.axis2_RaisePosition;
            executeAxis2Move := TRUE;
            
            IF fbMoveAxis2.Done THEN
                executeAxis2Move := FALSE;
                subStep := 2;
            END_IF;
            
            IF fbMoveAxis2.Error THEN
                executeAxis2Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        2: // Step 4: Axes 6,7 shift block toward center
            targetPosAxis6 := positions.axis6_ShiftPosition;
            executeAxis6Move := TRUE;
            
            IF fbMoveAxis6.Done THEN
                executeAxis6Move := FALSE;
                subStep := 3;
            END_IF;
            
            IF fbMoveAxis6.Error THEN
                executeAxis6Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        3: // Step 5: Axes 2,3,4,5 lower (back down below table)
            targetPosAxis2 := positions.axis2_LowerPosition;
            executeAxis2Move := TRUE;
            
            IF fbMoveAxis2.Done THEN
                executeAxis2Move := FALSE;
                subStep := 4;
            END_IF;
            
            IF fbMoveAxis2.Error THEN
                executeAxis2Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        4: // Step 6: Axes 6,7 return to home position
            targetPosAxis6 := positions.axis6_HomePosition;
            executeAxis6Move := TRUE;
            
            IF fbMoveAxis6.Done THEN
                executeAxis6Move := FALSE;
                subStep := 5;
            END_IF;
            
            IF fbMoveAxis6.Error THEN
                executeAxis6Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        5: // Step 7: Axes 10,11,12,13 raise (prepare to receive block)
            targetPosAxis10 := positions.axis10_RaisePosition;
            executeAxis10Move := TRUE;
            
            IF fbMoveAxis10.Done THEN
                executeAxis10Move := FALSE;
                subStep := 6;
            END_IF;
            
            IF fbMoveAxis10.Error THEN
                executeAxis10Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        6: // Step 8: Axes 8,9 shift block to center
            targetPosAxis8 := positions.axis8_ShiftPosition;
            executeAxis8Move := TRUE;
            
            IF fbMoveAxis8.Done THEN
                executeAxis8Move := FALSE;
                subStep := 7;
            END_IF;
            
            IF fbMoveAxis8.Error THEN
                executeAxis8Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        7: // Step 9: Wait for suction cups to enable
            // Enable suction cups output
            // suctionCupsEnable := TRUE;  // TODO: Add this output
            
            // Wait for confirmation that suction is holding
            // IF suctionCupsActive THEN  // TODO: Add this input
            //     subStep := 8;
            // END_IF;
            
            // For now, just proceed immediately
            subStep := 8;
        
        // ----------------------------------------------------------------
        8: // Step 10: Axes 8,9 return to home position
            targetPosAxis8 := positions.axis8_HomePosition;
            executeAxis8Move := TRUE;
            
            IF fbMoveAxis8.Done THEN
                executeAxis8Move := FALSE;
                subStep := 9;
            END_IF;
            
            IF fbMoveAxis8.Error THEN
                executeAxis8Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        9: // Step 11: Axes 10,11,12,13 lower (get out of way)
            targetPosAxis10 := positions.axis10_LowerPosition;
            executeAxis10Move := TRUE;
            
            IF fbMoveAxis10.Done THEN
                executeAxis10Move := FALSE;
                subStep := 10;
            END_IF;
            
            IF fbMoveAxis10.Error THEN
                executeAxis10Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        10: // Complete - block is now in center, ready for carving
            // Determine next operation based on recipe enable flags
            IF enableStudsOperation THEN
                motionStep := E_MotionStep.STUDS;
            ELSIF enableWindowsDoorsOperation THEN
                motionStep := E_MotionStep.WINDOWS_DOORS;
            ELSIF enableElectricalOperation THEN
                motionStep := E_MotionStep.ELECTRICAL;
            ELSE
                // All operations disabled - skip to unload
                motionStep := E_MotionStep.SHIFT_TO_UNLOAD_AREA;
            END_IF
            subStep := 0;

    END_CASE;
	
	E_MotionStep.STUDS:
    
    CASE subStep OF
        
        // ----------------------------------------------------------------
        0: // Step 12: Axes (14,16)(18,20)(22,24)(26,28) move to position 1
            // These are the vertical axes (even masters)
            targetPosAxis14 := positions.axis14_TargetPosition_1;
            targetPosAxis18 := positions.axis18_TargetPosition_1;
            targetPosAxis22 := positions.axis22_TargetPosition_1;
            targetPosAxis26 := positions.axis26_TargetPosition_1;
            
            executeAxis14Move := TRUE;
            executeAxis18Move := TRUE;
            executeAxis22Move := TRUE;
            executeAxis26Move := TRUE;
            
            // Wait for all to complete
            IF fbMoveAxis14.Done AND 
               fbMoveAxis18.Done AND 
               fbMoveAxis22.Done AND 
               fbMoveAxis26.Done THEN
                
                executeAxis14Move := FALSE;
                executeAxis18Move := FALSE;
                executeAxis22Move := FALSE;
                executeAxis26Move := FALSE;
                subStep := 1;
            END_IF;
            
            // Check for errors
            IF fbMoveAxis14.Error OR 
               fbMoveAxis18.Error OR 
               fbMoveAxis22.Error OR 
               fbMoveAxis26.Error THEN
                
                executeAxis14Move := FALSE;
                executeAxis18Move := FALSE;
                executeAxis22Move := FALSE;
                executeAxis26Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        1: // Step 13: Axes (15,17)(19,21)(23,25)(27,29) move to position 1
            // These are the horizontal axes (odd masters)
            targetPosAxis15 := positions.axis15_TargetPosition_1;
            targetPosAxis19 := positions.axis19_TargetPosition_1;
            targetPosAxis23 := positions.axis23_TargetPosition_1;
            targetPosAxis27 := positions.axis27_TargetPosition_1;
            
            executeAxis15Move := TRUE;
            executeAxis19Move := TRUE;
            executeAxis23Move := TRUE;
            executeAxis27Move := TRUE;
            
            // Wait for all to complete
            IF fbMoveAxis15.Done AND 
               fbMoveAxis19.Done AND 
               fbMoveAxis23.Done AND 
               fbMoveAxis27.Done THEN
                
                executeAxis15Move := FALSE;
                executeAxis19Move := FALSE;
                executeAxis23Move := FALSE;
                executeAxis27Move := FALSE;
                subStep := 2;
            END_IF;
            
            // Check for errors
            IF fbMoveAxis15.Error OR 
               fbMoveAxis19.Error OR 
               fbMoveAxis23.Error OR 
               fbMoveAxis27.Error THEN
                
                executeAxis15Move := FALSE;
                executeAxis19Move := FALSE;
                executeAxis23Move := FALSE;
                executeAxis27Move := FALSE;
                // TODO: Handle error
            END_IF;
        
        // ----------------------------------------------------------------
        2: // Step 14: Axes (14,16)(18,20)(22,24)(26,28) move to position 2
            // Vertical axes carving down/up
            targetPosAxis14 := positions.axis14_TargetPosition_2;
            targetPosAxis18 := positions.axis18_TargetPosition_2;
            targetPosAxis22 := positions.axis22_TargetPosition_2;
            targetPosAxis26 := positions.axis26_TargetPosition_2;
            
            executeAxis14Move := TRUE;
            executeAxis18Move := TRUE;
            executeAxis22Move := TRUE;
            executeAxis26Move := TRUE;
            
            // Wait for all to complete
            IF fbMoveAxis14.Done AND 
               fbMoveAxis18.Done AND 
               fbMoveAxis22.Done AND 
               fbMoveAxis26.Done THEN
                
                executeAxis14Move := FALSE;
                executeAxis18Move := FALSE;
                executeAxis22Move := FALSE;
                executeAxis26Move := FALSE;
                subStep := 3;
            END_IF;
            
            // Check for errors
            IF fbMoveAxis14.Error OR 
               fbMoveAxis18.Error OR 
               fbMoveAxis22.Error OR

                executeAxis22Move := FALSE;
                executeAxis23Move := FALSE;

    CASE subStep OF

            targetPosAxis30 := positions.axis30_TargetPosition_3;
