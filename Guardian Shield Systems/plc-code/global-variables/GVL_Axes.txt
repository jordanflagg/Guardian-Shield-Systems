{attribute 'qualified_only'}
VAR_GLOBAL CONSTANT
    // ============================================================================
    // MASTER AXIS CONFIGURATION
    // Single source of truth for all master axis definitions
    // ============================================================================
    // NOTE: Gantry groups (slave axis following) are configured in the Bosch ctrlX
    // web interface, not in PLC code. PLC only commands master axes directly.
    //
    // Axis Groups:
    //   1      - Shift left (independent)
    //   2      - Load table lift (gantry group: 2,3,4,5 in Bosch UI)
    //   6      - Shift center (gantry group: 6,7 in Bosch UI)
    //   8      - Pusher (gantry group: 8,9 in Bosch UI)
    //   10     - Unload table lift (gantry group: 10,11,12,13 in Bosch UI)
    //   14,15  - Hot wire cutting station 1 (gantry groups: 14+16, 15+17 in Bosch UI)
    //   18,19  - Hot wire cutting station 2 (gantry groups: 18+20, 19+21 in Bosch UI)
    //   22,23  - Hot wire cutting station 3 (gantry groups: 22+24, 23+25 in Bosch UI)
    //   26,27  - Hot wire cutting station 4 (gantry groups: 26+28, 27+29 in Bosch UI)
    //   30,31  - Hot wire cutting large station (gantry groups: 30+32, 31+33 in Bosch UI)
    //   34     - Reserved
    //   35,36  - Hot wire cutting independent
    // ============================================================================

    MASTER_AXES : ARRAY[1..18] OF INT := [
        1,      // Shift left
        2,      // Load table lift
        6,      // Shift center
        8,      // Pusher
        10,     // Unload table lift
        14,     // Hot wire station 1 Y-axis
        15,     // Hot wire station 1 X-axis
        18,     // Hot wire station 2 Y-axis
        19,     // Hot wire station 2 X-axis
        22,     // Hot wire station 3 Y-axis
        23,     // Hot wire station 3 X-axis
        26,     // Hot wire station 4 Y-axis
        27,     // Hot wire station 4 X-axis
        30,     // Hot wire large station X-axis
        31,     // Hot wire large station Y-axis
        34,     // Reserved
        35,     // Hot wire independent 1
        36      // Hot wire independent 2
    ];

    MASTER_AXIS_COUNT : INT := 18;
    TOTAL_AXIS_COUNT : INT := 36;

END_VAR
