(*
==============================================================================
  GVL_GCode
  Global Variable List - G-Code Script Runtime Interface

  The Python pipeline writes a G-Code file directly to the ctrlX filesystem
  and triggers execution via a POST command to the Data Layer. The PLC monitors
  the script instance state to determine when cutting is complete or faulted.

  Script instance state is read from:
    script/instances//<sScriptInstance>/state

  State machine (from Bosch ctrlX documentation):
    INIT       - Instance created, ready to run
    BUSY       - Script actively executing (G-Code running kinematics)
    READY      - Run-string completed normally
    RESTART    - Exception occurred, awaiting restart/reset
    TERMINATED - After restart/reset or shutdown
    FINISHED   - After TERMINATED via shutdown
    ERROR      - Script error
    ABORTING   - Abort command received, transitioning back to INIT

  For file execution (Python POST to run file):
    INIT → BUSY → INIT   (normal completion)
    INIT → BUSY → ERROR  (fault)
    INIT → BUSY → ABORTING → INIT  (abort)

  Diagnostic info available at:
    script/instances//<sScriptInstance>/diag
==============================================================================
*)

{attribute 'qualified_only'}
VAR_GLOBAL
    // ═══════════════════════════════════════════════════════════════════════
    // SCRIPT INSTANCE CONFIGURATION
    // Must match the instance name configured in ctrlX script runtime
    // ═══════════════════════════════════════════════════════════════════════

    sScriptInstance : STRING[64] := 'cutting';      // Instance name in ctrlX runtime

    // ═══════════════════════════════════════════════════════════════════════
    // SCRIPT STATE (Written by FB_MotionStep_Cutting via Data Layer read)
    // Values: INIT / BUSY / READY / RESTART / TERMINATED / FINISHED / ERROR / ABORTING
    // ═══════════════════════════════════════════════════════════════════════

    sScriptState : STRING[32];                      // Current state from Data Layer

    // ═══════════════════════════════════════════════════════════════════════
    // HANDSHAKE BITS (PLC writes, Python reads via Data Layer subscription)
    // ═══════════════════════════════════════════════════════════════════════

    // PLC sets this TRUE when it wants Python to POST to the G-Code runtime.
    // Python sees it, fires the POST, then clears it (sets FALSE).
    // PLC also clears it once it confirms the script is BUSY (belt-and-suspenders).
    bTriggerStart : BOOL;   // PLC → Python: "please start the cutting job now"

    // Optional: Python sets TRUE to acknowledge it received the trigger and fired the POST.
    // PLC can use this to distinguish "trigger sent but not yet seen" from "POST sent".
    bStartAck     : BOOL;   // Python → PLC: "POST has been sent"

    // ═══════════════════════════════════════════════════════════════════════
    // STATUS FLAGS (Derived from sScriptState by FB_MotionStep_Cutting)
    // ═══════════════════════════════════════════════════════════════════════

    bBusy        : BOOL;    // TRUE while G-Code script is executing
    bJobDone     : BOOL;    // TRUE when script completed successfully
    bError       : BOOL;    // TRUE when script is in ERROR state

    // ═══════════════════════════════════════════════════════════════════════
    // HMI / DIAGNOSTIC
    // ═══════════════════════════════════════════════════════════════════════

    sStatus      : STRING[255];                     // Human-readable status for HMI
    sDiag        : STRING[255];                     // Diagnostic info (from /diag path)

END_VAR
