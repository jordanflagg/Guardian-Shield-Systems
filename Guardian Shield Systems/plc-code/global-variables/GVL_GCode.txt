(*
==============================================================================
  GVL_GCode
  Global Variable List — G-Code Pipeline Interface

  Architecture (v2.0 — Native PLC Dispatch):
    The Python API (launch_api.bat) processes STEP files, generates G-code,
    uploads Wire*.npg files to the ctrlX via WebDAV, then sets bFilesReady.

    FB_MotionStep_Cutting takes it from there — entirely within the PLC:
      1. Waits for bFilesReady = TRUE
      2. Prepares each wire kinematic (check state, reset+enable if needed)
      3. Resets each script instance
      4. Dispatches Wire1-Wire4 (Wire5 if bWire5Active) via DL_WriteNode
      5. Monitors kinematic opstate until all STANDBY (cutting done)
      6. Dispatches Wire*_home.npg to retract wires to X=0
      7. Waits for home moves to complete
      8. Clears bFilesReady, sets bJobDone, reports Done

  The trigger_dispatch.py Python service is no longer needed.
  The API sets nActiveWires and bWire5Active when staging files.

  Script file paths on ctrlX:
    Cutting : /var/snap/rexroth-solutions/common/solutions/
              activeConfiguration/scripts/gcode/Wire{N}.npg
    Home    : same folder / Wire{N}_home.npg
==============================================================================
*)

{attribute 'qualified_only'}
{attribute 'symbol' := 'readwrite'}
VAR_GLOBAL

    // ═══════════════════════════════════════════════════════════════════════
    // WIRE CONFIGURATION
    // Set by the Python API at staging time before bFilesReady goes TRUE.
    // ═══════════════════════════════════════════════════════════════════════

    // Wire name lookup — index matches FB_MotionStep_Cutting loop (1..5)
    WIRE_NAMES : ARRAY[1..5] OF STRING[20] := [
        'Wire1', 'Wire2', 'Wire3', 'Wire4', 'Wire5'
    ];

    // Base path for all .npg files on the ctrlX filesystem
    SCRIPT_BASE_PATH : STRING[120] :=
        '/var/snap/rexroth-solutions/common/solutions/activeConfiguration/scripts/gcode/';

    // Number of active wires this cycle (4 = stud cuts only, 5 = includes window/door)
    // Written by the Python API before setting bFilesReady.
    {attribute 'symbol' := 'readwrite'}
    nActiveWires : INT := 4;

    // TRUE when a Wire5 script (window/door opening) was staged this cycle.
    // Set by the Python API alongside nActiveWires.
    {attribute 'symbol' := 'readwrite'}
    bWire5Active : BOOL := FALSE;

    // ═══════════════════════════════════════════════════════════════════════
    // PIPELINE HANDSHAKE
    // ═══════════════════════════════════════════════════════════════════════

    // Python API → PLC: G-code files uploaded and ready to run.
    // FB_MotionStep_Cutting will not dispatch until this is TRUE.
    // Cleared by FB_MotionStep_Cutting when the cycle completes so the next
    // block requires a fresh pipeline upload.
    {attribute 'symbol' := 'readwrite'}
    bFilesReady : BOOL;

    // Pipeline progress stage — written by the Python API as it advances,
    // then updated by FB_MotionStep_Cutting during execution.
    // 0=Idle  1=STEP processing  2=G-code generated  3=Uploaded
    // 4=Ready to run  5=Cutting  6=Complete  99=Error
    {attribute 'symbol' := 'readwrite'}
    nPipelineStage : INT;

    // ═══════════════════════════════════════════════════════════════════════
    // STATUS FLAGS (written by FB_MotionStep_Cutting)
    // Readable by the Python API and HMI via Data Layer subscription.
    // ═══════════════════════════════════════════════════════════════════════

    {attribute 'symbol' := 'readwrite'}
    bBusy    : BOOL;    // TRUE while wires are cutting or homing

    {attribute 'symbol' := 'readwrite'}
    bJobDone : BOOL;    // TRUE when cutting + homing completed normally

    {attribute 'symbol' := 'readwrite'}
    bError   : BOOL;    // TRUE when a dispatch or kinematic error occurred

    // ═══════════════════════════════════════════════════════════════════════
    // HMI / DIAGNOSTIC
    // ═══════════════════════════════════════════════════════════════════════

    {attribute 'symbol' := 'readwrite'}
    sStatus : STRING[255];      // Human-readable status string for HMI display

END_VAR
