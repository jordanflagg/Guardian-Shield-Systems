FUNCTION FC_AnyMoverError : BOOL
// ============================================================================
// Checks if any of the provided movers have an error
// Returns TRUE if ANY mover has Error = TRUE
//
// Usage:
//   hasError := FC_AnyMoverError(
//       (Done := mover1.Done, Error := mover1.Error, AxisID := 14),
//       (Done := mover2.Done, Error := mover2.Error, AxisID := 18),
//       ...
//   );
//
// Empty slots (AxisID = 0) are ignored
// ============================================================================

VAR_INPUT
    Mover1 : ST_MoverStatus;
    Mover2 : ST_MoverStatus;
    Mover3 : ST_MoverStatus;
    Mover4 : ST_MoverStatus;
    Mover5 : ST_MoverStatus;
    Mover6 : ST_MoverStatus;
    Mover7 : ST_MoverStatus;
    Mover8 : ST_MoverStatus;
END_VAR

///////////////////////////////////////////////////////////////////////////////

FC_AnyMoverError := FALSE;

// Check each mover for errors
IF Mover1.AxisID > 0 AND Mover1.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover2.AxisID > 0 AND Mover2.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover3.AxisID > 0 AND Mover3.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover4.AxisID > 0 AND Mover4.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover5.AxisID > 0 AND Mover5.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover6.AxisID > 0 AND Mover6.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover7.AxisID > 0 AND Mover7.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
IF Mover8.AxisID > 0 AND Mover8.Error THEN FC_AnyMoverError := TRUE; RETURN; END_IF
