PROGRAM PLC_PRG
VAR
    // Function blocks
    buttonManager : FB_ButtonManager;
    axisPower : FB_AxisPower;
    stateMachine : FB_MachineStateMachine;

    // HMI Data Exchange - Called directly from PLC_PRG (not buried in FBs)
    axisData : FB_AxisData;             // Reads position/velocity for jog axis
    axisDataHome : FB_AxisData;         // Reads position/velocity for homing axis (separate instance)
    
    // Raw button inputs (you toggle these for testing)
    btn_Start : BOOL := FALSE;
    btn_Stop : BOOL := FALSE;
    btn_Home : BOOL := FALSE;
    btn_Manual : BOOL := FALSE;
    btn_Reset : BOOL := FALSE;
    btn_EStop : BOOL := FALSE;
    btn_DriveOn : BOOL := FALSE;

    // Machine state for HMI
    MACHINE_STATE : E_MachineState;

    // HMI Status Variables (read by web HMI)
    isRunning : BOOL;
    isFaulted : BOOL;
    isHomed : BOOL;

    // HMI Recipe Variables
    HMI_RecipeSelect : UINT := 1;           // Recipe dropdown selection (1-10)
    HMI_ActiveRecipe : UINT := 1;           // Currently active recipe ID
    HMI_RecipeName : STRING[40];            // Recipe name for display
    HMI_RecipeDescription : STRING[100];    // Recipe description for display
    HMI_RecipeLocked : BOOL := FALSE;       // TRUE when recipe cannot be changed
    HMI_EnableStuds : BOOL := TRUE;         // Shows if STUDS operation enabled
    HMI_EnableWindows : BOOL := TRUE;       // Shows if WINDOWS operation enabled
    HMI_EnableElectrical : BOOL := TRUE;    // Shows if ELECTRICAL operation enabled

    // HMI Motion Sequence Visualization Variables
    HMI_MotionStep : E_MotionStep;          // Current motion step
    HMI_MotionSubStep : INT;                // Current substep number
    HMI_ActiveAxesMask : LWORD;             // Bitmask of currently moving axes (64-bit for 36 axes)
    HMI_StepDescription : STRING[80];       // Human-readable description

    // HMI Error Variables
    HMI_HasError : BOOL := FALSE;           // TRUE if motion error occurred
    HMI_ErrorAxisID : UINT := 0;            // Which axis caused the error (0 = none)
    HMI_DrivesReady : BOOL := FALSE;        // TRUE when all drives are powered

    // HMI Manual Jog Control Variables (written by web HMI)
    HMI_JogAxis : UINT := 1;                // Selected axis for jogging (1-36, master axes only)
    HMI_JogDirection : BOOL := FALSE;       // TRUE = positive, FALSE = negative
    HMI_JogSpeed : LREAL := 10.0;           // Jog velocity in mm/s
    HMI_JogEnable : BOOL := FALSE;          // TRUE while jog button held down

    // HMI Manual Jog Feedback Variables (read by web HMI)
    HMI_JogPosition : LREAL := 0.0;         // Current position of selected axis
    HMI_JogVelocity : LREAL := 0.0;         // Current velocity of selected axis
    HMI_JogActive : BOOL := FALSE;          // TRUE when jog is actively moving
    HMI_JogError : BOOL := FALSE;           // TRUE if jog error occurred
    HMI_JogInvalidAxis : BOOL := FALSE;     // TRUE if selected axis is not a master axis

    // HMI Homing Control Variables (written by web HMI)
    HMI_HomeAxis : UINT := 1;               // Selected axis for homing (1-36, master axes only)
    HMI_HomeSingle : BOOL := FALSE;         // Trigger: home selected axis
    HMI_HomeAll : BOOL := FALSE;            // Trigger: home all master axes
    HMI_HomeSpeed : LREAL := 10.0;          // Homing velocity in mm/s

    // HMI Homing Feedback Variables (read by web HMI)
    HMI_HomeBusy : BOOL := FALSE;           // TRUE when homing operation in progress
    HMI_HomeDone : BOOL := FALSE;           // TRUE when homing operation completed
    HMI_HomeError : BOOL := FALSE;          // TRUE if homing error occurred
    HMI_HomeErrorAxisID : UINT := 0;        // Axis that caused homing error
    HMI_HomeInvalidAxis : BOOL := FALSE;    // TRUE if selected axis is not a master axis
    HMI_AllAxesHomed : BOOL := FALSE;       // TRUE when all master axes are at position 0
    HMI_AxisIsHomed : ARRAY[1..36] OF BOOL; // Individual axis homed status
    HMI_HomePosition : LREAL := 0.0;        // Current position of selected homing axis

    // HMI Loaded Recipe Tracking
    HMI_LoadedRecipeId : UINT := 1;         // Recipe currently loaded in PLC memory
END_VAR

/////////////////////////////////////////////////

// ============================================================================
// BUTTON MANAGEMENT - All logic encapsulated in FB
// ============================================================================
buttonManager(
    btn_Start_In := btn_Start,
    btn_Stop_In := btn_Stop,
    btn_Home_In := btn_Home,
    btn_Manual_In := btn_Manual,
    btn_Reset_In := btn_Reset,
    btn_EStop_In := btn_EStop
);

// ============================================================================
// AXIS POWER
// ============================================================================
// Automatically disable drives on fault/E-stop (user must re-enable manually)
IF stateMachine.isFaulted THEN
    btn_DriveOn := FALSE;
END_IF

axisPower(
    PowerOn := btn_DriveOn,
    Reset := buttonManager.btn_Reset_Out
);

// ============================================================================
// STATE MACHINE - Handles machine state logic only (no HMI data pass-through)
// ============================================================================
stateMachine(
    startButton := buttonManager.btn_Start_Out,
    stopButton := buttonManager.btn_Stop_Out,
    homeButton := buttonManager.btn_Home_Out,
    manualMode := buttonManager.btn_Manual_Out,
    eStopActive := buttonManager.btn_EStop_Out,
    resetButton := buttonManager.btn_Reset_Out,
    selectedRecipe := HMI_RecipeSelect,
);

MACHINE_STATE := stateMachine.currentStateOut;

// ============================================================================
// HMI AXIS DATA - Always reads position/velocity for selected axes
// Called directly from PLC_PRG, independent of machine state
// Separate instances for Manual (jog) and Homing pages to avoid interference
// ============================================================================
axisData(
    AxisSelect := HMI_JogAxis
);
axisDataHome(
    AxisSelect := HMI_HomeAxis
);

// Update HMI with real-time axis data (always available)
HMI_JogPosition := axisData.Position;
HMI_JogVelocity := axisData.Velocity;
HMI_HomePosition := axisDataHome.Position;



// Update HMI with jog feedback
HMI_JogActive := stateMachine.axisManual.Active;
HMI_JogError := stateMachine.axisManual.Error;
HMI_JogInvalidAxis := stateMachine.axisManual.InvalidAxis;

// Update HMI with homing feedback
HMI_HomeBusy := stateMachine.fbHome.Busy;
HMI_HomeDone := stateMachine.fbHome.Done;
HMI_HomeError := stateMachine.fbHome.Error;
HMI_HomeErrorAxisID := stateMachine.fbHome.ErrorAxisID;
HMI_HomeInvalidAxis := stateMachine.fbHome.InvalidAxis;
HMI_AllAxesHomed := stateMachine.fbHome.AllAxesHomed;
HMI_AxisIsHomed := stateMachine.fbHome.AxisIsHomed;

// ============================================================================
// UPDATE HMI STATUS VARIABLES
// ============================================================================
// Expose state machine status to web HMI
isRunning := stateMachine.isRunning;
isFaulted := stateMachine.isFaulted;
isHomed := stateMachine.isHomed;

// ============================================================================
// UPDATE RECIPE HMI VARIABLES
// ============================================================================
HMI_ActiveRecipe := stateMachine.activeRecipeID;
HMI_RecipeLocked := stateMachine.recipeLocked;

// Update recipe name and description from library
IF HMI_ActiveRecipe >= 1 AND HMI_ActiveRecipe <= 10 THEN
    HMI_RecipeName := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].recipeName;
    HMI_RecipeDescription := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].recipeDescription;
    HMI_EnableStuds := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].enableStuds;
    HMI_EnableWindows := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].enableWindowsDoors;
    HMI_EnableElectrical := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].enableElectrical;
ELSE
    HMI_RecipeName := 'INVALID';
    HMI_RecipeDescription := 'Recipe ID out of range';
    HMI_EnableStuds := FALSE;
    HMI_EnableWindows := FALSE;
    HMI_EnableElectrical := FALSE;
END_IF

// ============================================================================
// UPDATE MOTION SEQUENCE HMI VARIABLES
// ============================================================================
HMI_MotionStep := stateMachine.motionStepOut;
HMI_MotionSubStep := stateMachine.motionSubStep;
HMI_ActiveAxesMask := stateMachine.activeAxesMask;
HMI_StepDescription := stateMachine.stepDescription;

// ============================================================================
// UPDATE ERROR HMI VARIABLES
// ============================================================================
HMI_HasError := stateMachine.hasError;
HMI_ErrorAxisID := stateMachine.errorInfo.errorAxisID;
HMI_DrivesReady := axisPower.DrivesReady;

// Track loaded recipe (for now, same as active - could be different if editing)
HMI_LoadedRecipeId := HMI_ActiveRecipe;
