PROGRAM PLC_PRG
VAR
    // Function blocks
    buttonManager : FB_ButtonManager;
    axisPower : FB_AxisPower;
    stateMachine : FB_MachineStateMachine;
    // Note: motionSequence is embedded inside stateMachine, not needed here
    
    // Raw button inputs (you toggle these for testing)
    btn_Start : BOOL := FALSE;
    btn_Stop : BOOL := FALSE;
    btn_Home : BOOL := FALSE;
    btn_Manual : BOOL := FALSE;
    btn_Reset : BOOL := FALSE;
    btn_EStop : BOOL := FALSE;
    btn_DriveOn : BOOL := FALSE;

    // Machine state for HMI
    MACHINE_STATE : E_MachineState;

    // HMI Status Variables (read by web HMI)
    isRunning : BOOL;
    isFaulted : BOOL;
    isHomed : BOOL;

    // HMI Recipe Variables
    HMI_RecipeSelect : UINT := 1;           // Recipe dropdown selection (1-10)
    HMI_ActiveRecipe : UINT := 1;           // Currently active recipe ID
    HMI_RecipeName : STRING[40];            // Recipe name for display
    HMI_RecipeDescription : STRING[100];    // Recipe description for display
    HMI_RecipeLocked : BOOL := FALSE;       // TRUE when recipe cannot be changed
    HMI_EnableStuds : BOOL := TRUE;         // Shows if STUDS operation enabled
    HMI_EnableWindows : BOOL := TRUE;       // Shows if WINDOWS operation enabled
    HMI_EnableElectrical : BOOL := TRUE;    // Shows if ELECTRICAL operation enabled

    // HMI Motion Sequence Visualization Variables
    HMI_MotionStep : E_MotionStep;          // Current motion step
    HMI_MotionSubStep : INT;                // Current substep number
    HMI_ActiveAxesMask : LWORD;             // Bitmask of currently moving axes (64-bit for 36 axes)
    HMI_StepDescription : STRING[80];       // Human-readable description

    // HMI Error Variables
    HMI_HasError : BOOL := FALSE;           // TRUE if motion error occurred
    HMI_ErrorAxisID : UINT := 0;            // Which axis caused the error (0 = none)
    HMI_DrivesReady : BOOL := FALSE;        // TRUE when all drives are powered
END_VAR


/////////////////////////////////////////////////

// ============================================================================
// BUTTON MANAGEMENT - All logic encapsulated in FB
// ============================================================================
buttonManager(
    btn_Start_In := btn_Start,
    btn_Stop_In := btn_Stop,
    btn_Home_In := btn_Home,
    btn_Manual_In := btn_Manual,
    btn_Reset_In := btn_Reset,
    btn_EStop_In := btn_EStop
);

// ============================================================================
// AXIS POWER
// ============================================================================
// Automatically disable drives on fault/E-stop (user must re-enable manually)
IF stateMachine.isFaulted THEN
    btn_DriveOn := FALSE;
END_IF

axisPower(
    PowerOn := btn_DriveOn,
    Reset := buttonManager.btn_Reset_Out
);

// ============================================================================
// STATE MACHINE - Uses processed button outputs
// ============================================================================
stateMachine(
    startButton := buttonManager.btn_Start_Out,
    stopButton := buttonManager.btn_Stop_Out,
    homeButton := buttonManager.btn_Home_Out,
    manualMode := buttonManager.btn_Manual_Out,
    eStopActive := buttonManager.btn_EStop_Out,
    resetButton := buttonManager.btn_Reset_Out,
    selectedRecipe := HMI_RecipeSelect
);


MACHINE_STATE := stateMachine.currentStateOut;

// ============================================================================
// UPDATE HMI VARIABLES
// ============================================================================
// Expose state machine status to web HMI
isRunning := stateMachine.isRunning;
isFaulted := stateMachine.isFaulted;
isHomed := stateMachine.isHomed;

// ============================================================================
// UPDATE RECIPE HMI VARIABLES
// ============================================================================
HMI_ActiveRecipe := stateMachine.activeRecipeID;
HMI_RecipeLocked := stateMachine.recipeLocked;

// Update recipe name and description from library
IF HMI_ActiveRecipe >= 1 AND HMI_ActiveRecipe <= 10 THEN
    HMI_RecipeName := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].recipeName;
    HMI_RecipeDescription := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].recipeDescription;
    HMI_EnableStuds := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].enableStuds;
    HMI_EnableWindows := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].enableWindowsDoors;
    HMI_EnableElectrical := GVL_Recipes.RecipeLibrary[HMI_ActiveRecipe].enableElectrical;
ELSE
    HMI_RecipeName := 'INVALID';
    HMI_RecipeDescription := 'Recipe ID out of range';
    HMI_EnableStuds := FALSE;
    HMI_EnableWindows := FALSE;
    HMI_EnableElectrical := FALSE;
END_IF

// ============================================================================
// UPDATE MOTION SEQUENCE HMI VARIABLES
// ============================================================================
HMI_MotionStep := stateMachine.motionStepOut;
HMI_MotionSubStep := stateMachine.motionSubStep;
HMI_ActiveAxesMask := stateMachine.activeAxesMask;
HMI_StepDescription := stateMachine.stepDescription;

// ============================================================================
// UPDATE ERROR HMI VARIABLES
// ============================================================================
HMI_HasError := stateMachine.hasMotionError;
HMI_ErrorAxisID := stateMachine.errorAxisID;
HMI_DrivesReady := axisPower.DrivesReady;
