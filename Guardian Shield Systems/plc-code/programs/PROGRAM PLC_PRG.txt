PROGRAM PLC_PRG
VAR
    // Function blocks — execution order matters: faultMonitor first, then power, then state machine
    faultMonitor  : FB_FaultMonitor;        // ALWAYS runs first — fault detection before anything else
    buttonManager : FB_ButtonManager;
    axisPower     : FB_AxisPower;
    kinEnable     : FB_KinematicEnable;     // Runs every scan: 24/7 kin opstate monitor + cutting sequencer
    stateMachine  : FB_MachineStateMachine;

    // HMI Data Exchange - Called directly from PLC_PRG
    axisData     : FB_AxisData;             // Reads position/velocity for jog axis
    axisDataHome : FB_AxisData;             // Reads position/velocity for homing axis
END_VAR

/////////////////////////////////////////////////

// ============================================================================
// 1. FAULT MONITOR — runs unconditionally, always first
//    ctrlX diagnosis poll runs 24/7.
// ============================================================================
faultMonitor(
    Reset          := (GVL_HMI.MACHINE_STATE = E_MachineState.RESETTING),
    CurrentStep    := stateMachine.motionStepOut,
    CurrentSubStep := stateMachine.motionSubStep
);

// ============================================================================
// 2. BUTTON MANAGEMENT
// ============================================================================
buttonManager(
    btn_Start_In  := GVL_HMI.btn_Start,
    btn_Stop_In   := GVL_HMI.btn_Stop,
    btn_Home_In   := GVL_HMI.btn_Home,
    btn_Manual_In := GVL_HMI.btn_Manual,
    btn_Reset_In  := GVL_HMI.btn_Reset,
    btn_EStop_In  := GVL_HMI.btn_EStop
);

// ============================================================================
// 3. AXIS POWER
//    Automatically disable drives on fault/E-stop
// ============================================================================
IF stateMachine.isFaulted THEN
    GVL_HMI.btn_DriveOn := FALSE;
END_IF

axisPower(
    PowerOn := GVL_HMI.btn_DriveOn
);

// ============================================================================
// 4. KINEMATIC ENABLE
//    Runs every scan — two independent roles:
//      Enable  : enables Wire1-4 and Wire5_6 kins on rising edge of DrivesReady.
//                Kins stay enabled for the entire session — never disabled between
//                cutting groups. Re-enables automatically after E-stop drive recovery.
//      Monitor : polls all 5 kin opstates via DL_ReadNode every ~2 s; reports
//                ERRORSTOP faults via FC_ReportFault 24/7.
// ============================================================================
kinEnable(
    DrivesReady         := axisPower.DrivesReady,
    InterruptKinematics := stateMachine.interruptKinematics OR GVL_HMI.HMI_DbgInterruptKins,
    ContinueKinematics  := stateMachine.continueKinematics  OR GVL_HMI.HMI_DbgContinueKins,
    DbgKinEnableNum     := GVL_HMI.HMI_DbgKinEnableNum,
    DbgKinDisableNum    := GVL_HMI.HMI_DbgKinDisableNum
);

// ============================================================================
// 5. STATE MACHINE
// ============================================================================
stateMachine(
    startButton := buttonManager.btn_Start_Out,
    stopButton  := buttonManager.btn_Stop_Out,
    homeButton  := buttonManager.btn_Home_Out,
    manualMode  := buttonManager.btn_Manual_Out,
    eStopActive := buttonManager.btn_EStop_Out,
    resetButton := buttonManager.btn_Reset_Out,
    HasFault    := faultMonitor.HasFault
);

GVL_HMI.MACHINE_STATE := stateMachine.currentStateOut;

// ============================================================================
// 6. HMI AXIS DATA — always reads position/velocity for selected axes
// ============================================================================
axisData(AxisSelect := GVL_HMI.HMI_JogAxis);
axisDataHome(AxisSelect := GVL_HMI.HMI_HomeAxis);

GVL_HMI.HMI_JogPosition  := axisData.Position;
GVL_HMI.HMI_JogVelocity  := axisData.Velocity;
GVL_HMI.HMI_HomePosition := axisDataHome.Position;

// ============================================================================
// 7. UPDATE HMI VARIABLES
// ============================================================================

// Machine status
GVL_HMI.isRunning := stateMachine.isRunning;
GVL_HMI.isFaulted := stateMachine.isFaulted;
GVL_HMI.isHomed   := stateMachine.isHomed;

// Motion sequence visualization
GVL_HMI.HMI_MotionStep      := stateMachine.motionStepOut;
GVL_HMI.HMI_MotionSubStep   := stateMachine.motionSubStep;
GVL_HMI.HMI_ActiveAxesMask  := stateMachine.activeAxesMask;
GVL_HMI.HMI_StepDescription := stateMachine.stepDescription;

// Fault status
GVL_HMI.HMI_HasFault    := faultMonitor.HasFault;
GVL_HMI.HMI_ActiveFault := faultMonitor.ActiveFault;

// ctrlX diagnostics (for HMI badge and informational display)
GVL_HMI.HMI_DiagErrorCount := faultMonitor.DiagErrorCount;
GVL_HMI.HMI_DiagWarnings   := faultMonitor.DiagWarnCount;

// Drive status
GVL_HMI.HMI_DrivesReady := axisPower.DrivesReady;
GVL_HMI.HMI_CyclePaused := stateMachine.cyclePausedOut;

// Jog feedback
GVL_HMI.HMI_JogActive      := stateMachine.axisManual.Active;
GVL_HMI.HMI_JogError       := stateMachine.axisManual.Error;
GVL_HMI.HMI_JogInvalidAxis := stateMachine.axisManual.InvalidAxis;

// Homing feedback
GVL_HMI.HMI_HomeBusy        := stateMachine.fbHome.Busy;
GVL_HMI.HMI_HomeDone        := stateMachine.fbHome.Done;
GVL_HMI.HMI_HomeError       := stateMachine.fbHome.Error;
GVL_HMI.HMI_HomeErrorAxisID := stateMachine.fbHome.ErrorAxisID;
GVL_HMI.HMI_HomeInvalidAxis := stateMachine.fbHome.InvalidAxis;
GVL_HMI.HMI_AllAxesHomed    := stateMachine.fbHome.AllAxesHomed;
GVL_HMI.HMI_AxisIsHomed     := stateMachine.fbHome.AxisIsHomed;
